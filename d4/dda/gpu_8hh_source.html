327 waitReady();

564 // An oversized node's header stores size >> c_oversizedShift, and bit 0 of the link pointing at
565 // it flags it. That link also carries bit 23, so the DMA engine stops there and never reads the
566 // node; the engine ignores the low two bits of a link while MADR keeps them, so the interrupt
567 // handler sees the flag. tests/dma/dma.c covers that in linked_dma_800001_terminator and
568 // linked_dma_odd_terminator.
569 static constexpr unsigned c_oversizedShift = 4;
570 static constexpr unsigned c_maxNodeWords = 255;
571 static_assert((1u << c_oversizedShift) == c_oversizedGranularity);
