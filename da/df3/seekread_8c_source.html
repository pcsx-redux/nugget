100 static int firstDataHeaderIs(const SeekReadEvent* events, unsigned count, uint8_t s, uint8_t f) {
101 for (unsigned i = 0; i < count; i++) {
102 // The seek must not complete before the read delivers data.
103 if (events[i].cause == 2) return 0;
104 if (events[i].cause != 1) continue;
105 return (events[i].header[0] == 0) && (events[i].header[1] == s) && (events[i].header[2] == f);
106 }
107 return 0;
108 }
