164 // Sends Test with one subfunction byte and records the response.
165 // Whether the response is exactly one of the strings in the list.
166 static int cdlTestKnownString(const char* const* list, unsigned count, const uint8_t* response, uint8_t size) {
167 for (unsigned i = 0; i < count; i++) {
168 unsigned len = 0;
169 while (list[i][len]) len++;
170 if (len == size && __builtin_memcmp(list[i], response, size) == 0) return 1;
171 }
172 return 0;
173 }

193CESTER_TEST(cdlTestRegionAndChip, test_instance,
194 int resetDone = resetCDRom();
195 if (!resetDone) {
196 cester_assert_true(resetDone);
197 return;
198 }
199
200 // 22h answers the region string and 23h/24h the servo and signal processor chips, as
201 // plain text with no status byte. A SCPH-9002 (vC3) answers "for Europe" and "CXD2940Q".
202 // psx-spx: vC0 controllers do not have these subfunctions and answer error 11h 10h; only
203 // the 10h is checked, as no vC0 console has been measured.
204 static const char* regions[] = {"for Europe", "for U/C", "for Japan", "for NETNA", "for US/AEP"};
205 static const char* servoChips[] = {"CXD2940Q", "CXD1817Q", "CXD2545Q", "CXD1782BR"};
206 static const char* signalChips[] = {"CXD2940Q", "CXD1817Q", "CXD2545Q", "CXD2510Q"};
207 uint8_t cause, size, response[16];
208 uint32_t time;
209
210 // The version is the last byte of the Test 20h answer.
211 cester_assert_true(cdlTestSub(0x20, &cause, response, &size, &time));
212 cester_assert_uint_eq(4, size);
213 if (size != 4) return;
214 uint8_t version = response[3];
215
216 for (uint8_t sub = 0x22; sub <= 0x24; sub++) {
217 cester_assert_true(cdlTestSub(sub, &cause, response, &size, &time));
218 if (version == 0xc0) {
219 cester_assert_uint_eq(5, cause);
220 cester_assert_uint_eq(2, size);
221 cester_assert_uint_eq(0x10, response[1]);
222 continue;
223 }
224 cester_assert_uint_eq(3, cause);
225 if (sub == 0x22) {
226 cester_assert_true(cdlTestKnownString(regions, sizeof(regions) / sizeof(regions[0]), response, size));
227 } else if (sub == 0x23) {
228 cester_assert_true(cdlTestKnownString(servoChips, sizeof(servoChips) / sizeof(servoChips[0]), response, size));
229 } else {
230 cester_assert_true(cdlTestKnownString(signalChips, sizeof(signalChips) / sizeof(signalChips[0]), response, size));
231 }
232 }
