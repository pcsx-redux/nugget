27// AdvancedPad polling across a psyqo MemoryCardFileSystem write, against the
28// card in MEMCARD_PORT (0 = slot 1, 1 = slot 2). THIS TEST ERASES THAT CARD,
29// and, like tests/sio/memcard, refuses to touch a card holding a save it did
30// not write.
31//
32// It writes a multi-block file with the asynchronous writeFile while the frame
33// loop keeps running, and counts the pad polls that completed meanwhile
34// (AdvancedPad::pollCount). The filesystem holds the SIO0 bus for the whole
35// write, but the card is deselected between sector transfers, so the pad must
36// keep polling: at least one poll for every two frames of the write, and never
37// more than two frames in a row without one. The file
38// is then read back and compared byte for byte, so a poll that corrupted a
39// transfer fails the test too.
40//
41// Grading: every check prints PASS or FAIL, and the run ends on
42// "memcard-pad port N: all K checks passed" only when every check passed.
43

114// The file is named with the prefix tests/sio/memcard uses, so either test
115// accepts a card the other one left behind.
116constexpr const char kPrefix[] = "BIMEMTEST";
117constexpr const char kName[] = "BIMEMTEST-PAD";
118constexpr const char kTitle[] = "MT MEMCARD PAD TEST";
119// Two blocks: the title and icon frames plus 9000 bytes do not fit in one.
120constexpr uint32_t kLen = 9000;
121uint8_t s_payload[kLen];
122uint8_t s_readback[kLen];
123alignas(4) uint8_t s_frame[128];
124FS::Icon s_icon;
125
126bool startsWith(const uint8_t *s, const char *prefix) {
127 for (uint32_t i = 0; prefix[i]; i++) {
128 if (s[i] != static_cast<uint8_t>(prefix[i])) return false;
129 }
130 return true;
131}
132
133// Blank, or a Sony filesystem whose only saves carry kPrefix. A test card a
134// previous run of tests/sio/memcard left with a title frame over frame 0 is
135// refused here; that test's own run formats it.
136bool cardIsOursToErase() {
137 Error e = app.m_card.readSectorBlocking(kPort, 0, s_frame);
138 if (e != Error::OK) {
139 ramsyscall_printf("REFUSE: frame 0 read failed: %s\n", MC::errorMessage(e));
140 return false;
141 }
142 bool zeros = true, ones = true;
143 for (unsigned i = 0; i < 128; i++) {
144 zeros = zeros && s_frame[i] == 0x00;
145 ones = ones && s_frame[i] == 0xff;
146 }
147 if (zeros || ones) return true;
148 if (s_frame[0] != 'M' || s_frame[1] != 'C') {
149 ramsyscall_printf("REFUSE: frame 0 is neither blank nor a Sony header\n");
150 return false;
151 }
152 for (uint16_t s = 1; s <= 15; s++) {
153 e = app.m_card.readSectorBlocking(kPort, s, s_frame);
154 if (e != Error::OK) {
155 ramsyscall_printf("REFUSE: directory frame %d read failed: %s\n", s, MC::errorMessage(e));
156 return false;
157 }
158 if (s_frame[0] == 0x51 && !startsWith(s_frame + 0x0a, kPrefix)) {
159 ramsyscall_printf("REFUSE: block %d holds a save this test did not write\n", s);
160 return false;
161 }
162 }
163 return true;
164}

166[[noreturn]] void finish() {
167 if (s_failures == 0) {
168 ramsyscall_printf("memcard-pad port %d: all %d checks passed\n", MEMCARD_PORT, s_checks);
169 } else {
170 ramsyscall_printf("memcard-pad port %d: %d of %d checks FAILED\n", MEMCARD_PORT, s_failures, s_checks);
171 }
172 int code = s_failures == 0 ? 0 : 1;
173 pcsx_exit(code);
174 // On hardware: the break the farm reads back as the verdict, exit code in
175 // $a0, as tests/runtime/runtime.c does.
176 register int a0 asm("$4") = code;
177 __asm__ volatile("break 4, 0\n" : : "r"(a0) : "memory");
178 while (1);
179}

196void TestScene::frame() {
197 auto &pad = app.m_pad;
198 switch (m_state) {
199 case 0: {
200 ramsyscall_printf("psyqo memory card + pad, port %d (slot %d)\n", MEMCARD_PORT, MEMCARD_PORT + 1);
201 if (!check(app.m_card.probeBlocking(kPort) == Error::OK, "probe: a card answers") ||
202 !cardIsOursToErase()) {
203 s_failures++;
204 finish();
205 }
206 if (!checkError(app.m_fs.formatBlocking(app.gpu(), kPort), Error::OK, "format")) finish();
207 for (uint32_t i = 0; i < kLen; i++) s_payload[i] = static_cast<uint8_t>(i * 7 + (i >> 8) * 13 + 0x5a);
208 __builtin_memset(&s_icon, 0, sizeof(s_icon));
209 s_icon.frameCount = 1;
210 for (unsigned i = 0; i < 16; i++) s_icon.clut[i] = static_cast<uint16_t>(i * 0x0842);
211 for (unsigned i = 0; i < 128; i++) s_icon.pixels[0][i] = static_cast<uint8_t>(i);
212 m_frames = 0;
213 m_polls0 = pad.pollCount();
214 m_state = 1;
215 break;
216 }
217 case 1:
218 // Idle baseline: with no card operation the pad polls every frame.
219 if (++m_frames < 60) break;
220 {
221 uint32_t polls = pad.pollCount() - m_polls0;
222 ramsyscall_printf("baseline: %d polls in %d frames\n", polls, m_frames);
223 check(polls + 2 >= m_frames, "baseline: the pad polls every frame with no card operation");
224 }
225 m_frames = 0;
226 m_polls0 = pad.pollCount();
227 m_lastPolls = m_polls0;
228 m_state = 2;
229 app.m_fs.writeFile(kPort, kName, kTitle, s_icon, s_payload, kLen, [this](Error e) {
230 m_writeError = e;
231 m_writeDone = true;
232 });
233 break;
234 case 2: {
235 m_frames++;
236 uint32_t polls = pad.pollCount();
237 if (polls == m_lastPolls) {
238 if (++m_gap > m_maxGap) m_maxGap = m_gap;
239 } else {
240 m_gap = 0;
241 m_lastPolls = polls;
242 }
243 if (!m_writeDone) {
244 if (m_frames > 60 * 60) {
245 check(false, "write: finished within 60 s");
246 finish();
247 }
248 break;
249 }
250 m_writeFrames = m_frames;
251 m_writePolls = pad.pollCount() - m_polls0;
252 m_state = 3;
253 break;
254 }
255 case 3: {
256 checkError(m_writeError, Error::OK, "write: async writeFile");
257 ramsyscall_printf("write: %d polls in %d frames\n", m_writePolls, m_writeFrames);
258 check(m_writePolls * 2 >= m_writeFrames, "write: the pad polled at least every other frame");
259 ramsyscall_printf("write: longest run without a poll: %d frames\n", m_maxGap);
260 check(m_maxGap <= 2, "write: never more than two frames in a row without a poll");
261 uint32_t got = 0;
262 __builtin_memset(s_readback, 0, sizeof(s_readback));
263 checkError(app.m_fs.readFileBlocking(app.gpu(), kPort, kName, s_readback, kLen, &got), Error::OK,
264 "read back: readFile");
265 bool same = got == kLen;
266 for (uint32_t i = 0; same && i < kLen; i++) same = s_readback[i] == s_payload[i];
267 ramsyscall_printf("read back: %d bytes\n", got);
268 check(same, "read back: every byte matches what was written");
269 checkError(app.m_fs.formatBlocking(app.gpu(), kPort), Error::OK, "cleanup: format");
270 finish();
271 }
272 }
