386// ==========================================================================
387// 32bit control registers: read-back fidelity, and whether commands write them
388//
389// psx-spx leaves an unanswered "(Input?, R/W?)" on TR (cnt5-7), BK
390// (cnt13-15), FC (cnt21-23), screen offset (cnt24-25) and DQB (cnt28). The
391// sign-extension tests above cover the 16bit control registers only.
392//
393// The assertion is that all-ones does not read back as zero. A "two reads
394// agree" check does not work here: it passes on a register stuck at zero.
395// ==========================================================================
396
397#define GTE_CTRL32_EACH(M) \
398 M(5) M(6) M(7) M(13) M(14) M(15) M(21) M(22) M(23) M(24) M(25) M(28)
399
400#define GTE_CTRL32_PROBE(r) \
401 { \
402 uint32_t _a; \
403 cop2_putc(r, _pat); \
404 cop2_getc(r, _a); \
405 if (_pat == 0xffffffffu && _a == 0u) _responds = 0; \
406 ramsyscall_printf(" c%d=0x%08x", (r), _a); \
407 }
408
409#define GTE_CTRL32_SET(r) cop2_putc(r, 0x0bad0000u | (r));
410
411#define GTE_CTRL32_CHECK(r) \
412 { \
413 uint32_t _o; \
414 cop2_getc(r, _o); \
415 if (_o != (0x0bad0000u | (r))) { \
416 _changed++; \
417 ramsyscall_printf("CTRL32 %s wrote c%d: 0x%08x\n", _nm, (r), _o); \
418 } \
419 }
420
421// FLAG is the one control register that is an output. Seed it with a pattern no
422// command can reproduce (all-ones reads back as 0xfffff000) and check the seed is
423// gone afterwards: that is what "the command rewrites FLAG" means, and a command
424// whose own result is zero would be invisible to an "it changed" test.
425#define GTE_CTRL32_CMDPROBE(NAME, OP) \
426 do { \
427 const char *_nm = (NAME); \
428 uint32_t _flag; \
429 GTE_CTRL32_EACH(GTE_CTRL32_SET) \
430 cop2_putc(31, 0xffffffffu); \
431 cop2_cmd(OP); \
432 cop2_getc(31, _flag); \
433 if (_flag == 0xfffff000u) { \
434 _flagkept++; \
435 ramsyscall_printf("CTRL32 %s left FLAG seeded\n", _nm); \
436 } \
437 ramsyscall_printf("CTRL32 %s FLAG=0x%08x\n", _nm, _flag); \
438 GTE_CTRL32_EACH(GTE_CTRL32_CHECK) \
439 } while (0)
