167 unsigned count;
168
169 for (count = 0; count < 32; count++) {
170 waitCDRomIRQ();
171 uint8_t cause = ackCDRomCause();
172 causes[count] = cause;
173 responseSizes[count] = readResponse(responses + count * 16);
174 if (cause != 1) break;

176
177 for (unsigned i = 0; i < count; i++) {
178 cester_assert_uint_eq(1, causes[i]);

189 cester_assert_uint_lt(count, 17);
190 if (count >= 15) {
191 unsigned offset = count - 15;
192 unsigned upCount = 0;

194 // stat
195 cester_assert_uint_eq(0x82, responses[i * 16 + 0]);
196 // track
197 cester_assert_uint_eq(0x25, responses[i * 16 + 1]);
198 // index

200 unsigned index = i - offset;

202 // absolute minute time
