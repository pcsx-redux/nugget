37// Line classes for the gpu-timing sweep. The harness owns timing, the
38// baseline, the GPU state (E1h: blend mode, dither) and the drawn check; the
39// builder below only turns a class and a size into GP0 words.
40#define LINE_GOURAUD 1u // GP0(50h) family instead of GP0(40h)
41#define LINE_POLY 2u // polyline (48h/58h) instead of a single line
42#define LINE_SEMI 4u // semi-transparent (bit 25 of the command word)
43#define LINE_SLANTED 8u // 2:1 slanted segments instead of axis-aligned ones
44#define LINE_VERTICAL 16u // y-major segments instead of x-major ones
45
46#define LINE_MAX_SEGS 16u
47#define LINE_MAX_WORDS (2u * (LINE_MAX_SEGS + 1u) + 1u)

49// Writes one line command of class |kind| into |out| and returns the number
50// of words written, polyline terminator included. Returns 0 for a kind it
51// does not build.
52//
53// |len| is the length of each segment in pixels along its major axis,
54// 1..255. |segs| is the segment count: 1 for a single line, 2..LINE_MAX_SEGS
55// for a polyline.
56//
57// Geometry. A vertex word is (y << 16) | x.
58// - A single line runs from (0, 0) to (dx, dy), and a polyline bounces back
59// and forth between (0, 0) and (dx, dy).
60// - (dx, dy) is (len, 0) by default: horizontal.
61// - LINE_SLANTED: (len, len / 2), x-major with a 2:1 slope.
62// - LINE_VERTICAL swaps the axes: (0, len) vertical, and with LINE_SLANTED
63// (len / 2, len), y-major with a 1:2 slope.
64//
65// Contract the harness relies on:
66// - the first vertex is (0, 0), so pixel (0, 0) is drawn;
67// - every vertex lies inside 0..255 on both axes;
68// - a polyline ends with the word 55555555h;
69// - the colour words are non-black, so a drawn pixel differs from the
70// cleared background;
71// - the return value counts the polyline terminator.
72unsigned buildLineCommand(uint32_t* out, unsigned kind, unsigned len, unsigned segs);
