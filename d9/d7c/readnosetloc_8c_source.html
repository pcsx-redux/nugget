29// A read command sent with no new Setloc since the previous read: once while
30// the previous read is still delivering sectors, and once after a Pause. Logs
31// every interrupt, with the sector header for each data-ready one, and checks
32// where the second read starts.

35 typedef struct {
36 uint32_t time;
37 uint8_t cause;
38 uint8_t status;
39 uint8_t header[4];
40 } NoSetLocEvent;

42 // Collects interrupts until `wantData` data-ready ones were seen, the log
43 // is full, or 3s pass without an interrupt. Returns the new event count.
44 static unsigned noSetLocCollect(NoSetLocEvent* events, unsigned count, unsigned max, unsigned wantData) {
45 unsigned dataReady = 0;
46 while ((count < max) && (dataReady < wantData)) {
47 uint32_t timeout = 3000000;
48 if (!waitCDRomIRQWithTimeout(&timeout)) break;
49 NoSetLocEvent* e = &events[count++];
50 e->time = timeout;
51 e->cause = ackCDRomCause();
52 uint8_t response[16];
53 readResponse(response);
54 e->status = response[0];
55 e->header[0] = e->header[1] = e->header[2] = e->header[3] = 0;
56 if (e->cause == 1) {
57 CDROM_REG0 = 0;
58 CDROM_REG3 = 0x80;
59 while ((CDROM_REG0 & 0x40) == 0);
60 for (unsigned i = 0; i < 4; i++) e->header[i] = CDROM_REG2;
61 CDROM_REG3 = 0;
62 dataReady++;
63 }
64 }
65 return count;
66 }

68 // Pauses and consumes every interrupt up to the Pause completion, so that
69 // nothing is left pending for the next test. Returns 1 once INT2 is seen.
70 static int noSetLocPauseAndDrain() {
71 CDROM_REG0 = 0;
72 CDROM_REG1 = CDL_PAUSE;
73 while (1) {
74 uint32_t timeout = 3000000;
75 if (!waitCDRomIRQWithTimeout(&timeout)) return 0;
76 uint8_t cause = ackCDRomCause();
77 discardResponse();
78 if (cause == 2) return 1;
79 }
80 }

82 static void printNoSetLocEvents(const char* name, const NoSetLocEvent* events, unsigned count, unsigned split) {
83 ramsyscall_printf("%s, %u events:", name, count);

96 static uint32_t noSetLocHeaderLBA(const NoSetLocEvent* e) {
97 return MSF2LBA(btoi(e->header[0]), btoi(e->header[1]), btoi(e->header[2]));
98 }

100 // LBA of the last data-ready header in [from, to), or 0 if there is none.
101 static uint32_t noSetLocLastLBA(const NoSetLocEvent* events, unsigned from, unsigned to) {
102 uint32_t lba = 0;
103 for (unsigned i = from; i < to; i++) {
104 if (events[i].cause == 1) lba = noSetLocHeaderLBA(&events[i]);
105 }
106 return lba;
107 }
108
109 // LBA of the first data-ready header in [from, to), or 0 if there is none.
110 static uint32_t noSetLocFirstLBA(const NoSetLocEvent* events, unsigned from, unsigned to) {
111 for (unsigned i = from; i < to; i++) {
112 if (events[i].cause == 1) return noSetLocHeaderLBA(&events[i]);
113 }
114 return 0;
115 }
