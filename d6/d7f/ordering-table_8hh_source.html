86 template <Fragment Frag>
87 void insert(Frag& frag, int32_t z) {
88 // Ordering tables sort individual polygons, so a packet large enough to need the oversized
89 // encoding is nonsense here and deliberately unsupported. Catch it at the definition site
90 // instead: past the chain entry, sizeof(Frag) covers the payload (plus count, on the fixed
91 // variants), so it is a conservative upper bound, and it costs nothing at runtime.
92 static_assert((sizeof(Frag) - sizeof(Fragments::ChainEntry)) / sizeof(uint32_t) <= 255,
93 "Fragment too big to be inserted into an ordering table");
94 auto* table = m_table + 1;
95 if constexpr (safety == Safe::Yes) {
96 z = eastl::clamp(z, int32_t(0), int32_t(N - 1));
97 }
98#ifdef PS1_PC_PORT
99 frag.set(table[z].next, frag.getActualFragmentSize());
100 table[z].set(&frag, 0);
101#else
102 frag.set(&table[z], frag.getActualFragmentSize());
103 table[z].head = reinterpret_cast<uint32_t>(&frag) & 0xffffff;
104#endif
105 }
106
107 // NOTE: can't use from other classes (PCGPU) otherwise
108 psyqo::Fragments::ChainEntry m_table[N + 1];
109 private:
110 friend class GPU;
111};
