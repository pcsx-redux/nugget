32/* Bufferless: PCwrite streams the target's bytes straight out of its own buffer,
33 and PCread decodes the reply straight into the target's own buffer. No staging
34 RAM, no memcpy, no per-frame size cap (the target address is the buffer). */
35
36/* op and all scalar fields are u32 = two words, low half first (section 13:
37 PCDRV frames are 32-bit aligned). */
38static void putU32(uint32_t v) {
39 transportSendWord((uint16_t)(v & 0xffff));
40 transportSendWord((uint16_t)(v >> 16));
41}
42
43static uint32_t getU32(void) {
44 uint32_t lo = transportRecvWord();
45 uint32_t hi = transportRecvWord();
46 return lo | (hi << 16);
47}

49/* Byte length of a NUL-terminated PS1 string, counting the terminator. */
50static uint32_t ps1StrLenZ(uint32_t ptr) {
51 volatile const uint8_t *s = (volatile const uint8_t *)ptr;
52 uint32_t n = 0;
53 while (s[n]) n++;
54 return n + 1; /* include the NUL */
55}
56
57/* Stream a NUL-terminated PS1 string as words, low-byte-first, zero-padded to a
58 word boundary (section 2). lenZ counts the terminator. */
59static void putCstr(uint32_t ptr, uint32_t lenZ) {
60 volatile const uint8_t *s = (volatile const uint8_t *)ptr;
61 uint32_t words = (lenZ + 1) >> 1;
62 for (uint32_t w = 0; w < words; w++) {
63 uint32_t i = 2 * w;
64 uint8_t b0 = (i < lenZ) ? s[i] : 0;
65 uint8_t b1 = (i + 1 < lenZ) ? s[i + 1] : 0;
66 transportSendWord((uint16_t)(b0 | (b1 << 8)));
67 }
68}

70/* Receive a scalar PCDRV_RESP [op:u32][ret:s32] and return ret. Drains any
71 trailing words so the framing stays aligned; a malformed frame yields -1. */
72static int32_t recvScalarResp(uint32_t op) {
73 uint16_t type, len;
74 transportRecvBegin(&type, &len);
75
76 uint32_t echo = (len >= 2) ? getU32() : 0;
77 int32_t ret = (len >= 4) ? (int32_t)getU32() : -1;
78 for (uint16_t i = (len >= 4) ? 4 : (len >= 2 ? 2 : 0); i < len; i++) transportRecvWord();
79
80 int rc = transportRecvEnd();
81 if (rc != TRANSPORT_OK || type != MON_PCDRV_RESP || echo != op) return -1;
82 return ret;
83}

85/* Bulk PCDRV payloads travel in frames of at most this many bytes, so every
86 frame stays under the stream LEN cap (design section 13). */
87#define PCDRV_CHUNK 8192
88
89/* Take `words` payload words into dst[off..nbytes), dropping any padding past
90 nbytes. Returns how many bytes landed. */
91/* Counts every byte below nbytes as arrived, stores only those below cap. */
92static uint32_t recvBytes(volatile uint8_t *dst, uint32_t off, uint32_t nbytes, uint32_t cap, uint16_t words) {
93 uint32_t landed = 0;
94 for (uint16_t i = 0; i < words; i++) {
95 uint16_t w = transportRecvWord();
96 uint32_t bi = off + (uint32_t)i * 2;
97 if (bi < nbytes) landed++;
98 if (bi + 1 < nbytes) landed++;
99 if (bi < cap) dst[bi] = (uint8_t)(w & 0xff);
100 if (bi + 1 < cap) dst[bi + 1] = (uint8_t)(w >> 8);
101 }
102 return landed;
103}

105static void putBytes(volatile const uint8_t *src, uint32_t n) {
106 for (uint32_t i = 0; i < n; i += 2) {
107 uint8_t b0 = src[i];
108 uint8_t b1 = (i + 1 < n) ? src[i + 1] : 0;
109 transportSendWord((uint16_t)(b0 | (b1 << 8)));
110 }
111}

113void monitorServicePcdrv(struct Registers *r, uint32_t op) {
114 uint32_t a0 = r->GPR.n.a0;
115 uint32_t a1 = r->GPR.n.a1;
116 uint32_t a2 = r->GPR.n.a2;
117 uint32_t a3 = r->GPR.n.a3;
118
119 switch (op) {
120 case PCDRV_INIT:
121 transportSendBegin(MON_PCDRV_REQ, 2);
122 putU32(op);
123 transportSendEnd();
124 r->GPR.n.v0 = (uint32_t)recvScalarResp(op);
125 break;
126
127 case PCDRV_CREAT: /* fallthrough: same wire shape, different op */
128 case PCDRV_OPEN: {
129 /* REQ [op:u32][mode|flags:u32][name:cstr]; name = a0, mode/flags = a2. */
130 uint32_t lenZ = ps1StrLenZ(a0);
131 uint16_t cw = (uint16_t)((lenZ + 1) >> 1);
132 transportSendBegin(MON_PCDRV_REQ, (uint16_t)(4 + cw));
133 putU32(op);
134 putU32(a2);
135 putCstr(a0, lenZ);
136 transportSendEnd();
137 r->GPR.n.v0 = 0;
138 r->GPR.n.v1 = (uint32_t)recvScalarResp(op);
139 break;
140 }
141
142 case PCDRV_CLOSE:
143 /* REQ [op:u32][fd:u32]; fd = a0. */
144 transportSendBegin(MON_PCDRV_REQ, 4);
145 putU32(op);
146 putU32(a0);
147 transportSendEnd();
148 r->GPR.n.v0 = (uint32_t)recvScalarResp(op);
149 break;
150
151 case PCDRV_READ: {
152 /* REQ [op:u32][fd:u32][len:u32]; fd = a1, len = a2, buf = a3.
153 RESP [op:u32][ret:s32 = nbytes][bytes], at most PCDRV_CHUNK
154 bytes, then continuation RESP [op:u32][bytes] frames until
155 nbytes have arrived. Decode straight into the target's own
156 buffer - no staging. */
157 transportSendBegin(MON_PCDRV_REQ, 6);
158 putU32(op);
159 putU32(a1);
160 putU32(a2);
161 transportSendEnd();
162
163 volatile uint8_t *dst = (volatile uint8_t *)a3;
164 uint16_t type, len;
165 transportRecvBegin(&type, &len);
166 uint32_t echo = (len >= 2) ? getU32() : 0;
167 int32_t ret = (len >= 4) ? (int32_t)getU32() : -1;
168 uint32_t nbytes = (ret > 0) ? (uint32_t)ret : 0;
169 uint16_t consumed = (len >= 4) ? 4 : (len >= 2 ? 2 : 0);
170 uint32_t cap = nbytes < a2 ? nbytes : a2;
171 uint32_t got = recvBytes(dst, 0, nbytes, cap, len - consumed);
172 int ok = transportRecvEnd() == TRANSPORT_OK && type == MON_PCDRV_RESP && echo == op;
173 while (ok && got < nbytes) {
174 transportRecvBegin(&type, &len);
175 echo = (len >= 2) ? getU32() : 0;
176 got += recvBytes(dst, got, nbytes, cap, len >= 2 ? len - 2 : 0);
177 ok = transportRecvEnd() == TRANSPORT_OK && type == MON_PCDRV_RESP && echo == op && len > 2;
178 }
179 r->GPR.n.v0 = 0;
180 /* A host answering with more than was asked for is a protocol
181 error; the excess was drained, not stored. */
182 r->GPR.n.v1 = ok && nbytes <= a2 ? (uint32_t)ret : (uint32_t)-1;
183 break;
184 }
185
186 case PCDRV_WRITE: {
187 /* REQ [op:u32][fd:u32][len:u32][bytes], at most PCDRV_CHUNK bytes,
188 then continuation REQ [op:u32][bytes] frames until len bytes
189 have gone; fd = a1, len = a2, buf = a3. One RESP answers the
190 whole write. Stream the target's bytes straight out of its
191 buffer. */
192 if ((int32_t)a2 < 0) {
193 r->GPR.n.v0 = 0;
194 r->GPR.n.v1 = (uint32_t)-1;
195 break;
196 }
197 uint32_t total = a2;
198 volatile const uint8_t *src = (volatile const uint8_t *)a3;
199 uint32_t chunk = total < PCDRV_CHUNK ? total : PCDRV_CHUNK;
200 transportSendBegin(MON_PCDRV_REQ, (uint16_t)(6 + ((chunk + 1) >> 1)));
201 putU32(op);
202 putU32(a1);
203 putU32(total);
204 putBytes(src, chunk);
205 transportSendEnd();
206 for (uint32_t off = chunk; off < total; off += chunk) {
207 chunk = total - off < PCDRV_CHUNK ? total - off : PCDRV_CHUNK;
208 transportSendBegin(MON_PCDRV_REQ, (uint16_t)(2 + ((chunk + 1) >> 1)));
209 putU32(op);
210 putBytes(src + off, chunk);
211 transportSendEnd();
212 }
213 r->GPR.n.v0 = 0;
214 r->GPR.n.v1 = (uint32_t)recvScalarResp(op);
215 break;
216 }
217
218 case PCDRV_LSEEK:
219 /* REQ [op:u32][fd:u32][offset:s32][whence:u32]; fd = a0, offset = a2,
220 whence = a3. */
221 transportSendBegin(MON_PCDRV_REQ, 8);
222 putU32(op);
223 putU32(a0);
224 putU32(a2);
225 putU32(a3);
226 transportSendEnd();
227 r->GPR.n.v0 = 0;
228 r->GPR.n.v1 = (uint32_t)recvScalarResp(op);
229 break;
230
231 default:
232 /* Unknown PCDRV code: leave a failure in v0 rather than hanging. */
233 r->GPR.n.v0 = (uint32_t)-1;
234 break;
235 }
