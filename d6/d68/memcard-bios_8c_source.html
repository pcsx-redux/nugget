27// The BIOS memory card calls, against the card in MEMCARD_PORT (0 = slot 1,
28// 1 = slot 2): the low level sector calls _card_read / _card_write and the
29// "bu" device behind open / read / write / firstfile / erase / format. In the
30// emulator that is OpenBIOS; on a console it is the retail kernel, so the two
31// are held to the same expectations. THIS TEST ERASES THE CARD, after the same
32// check psyqo's memcard test makes that it holds nothing but this suite's own
33// files. Nothing here addresses the other port: _bu_init, which touches both,
34// is never called.
35//
36// Grading: every check prints PASS or FAIL, and the run ends on
37// "memcard-bios port N: all K checks passed" only when every check passed.

106// Which event ended the last low level operation, or -1 if none came.
107static int waitCard() {
108 for (uint32_t spin = 0; spin < 0x200000; spin++) {
109 for (int i = 0; i < EVENTS; i++) {
110 if (syscall_testEvent(s_events[i])) return i;
111 }
112 }
113 return -1;
114}

116static const char *eventName(int e) { return e < 0 ? "no event" : s_names[e]; }

118// A card reports itself new on every access until something writes to it,
119// and the kernel turns that into a "new card" event. _new_card tells the
120// kernel to let the next access through; the access is then retried once.
121static int cardRead(int sector, uint8_t *buffer) {
122 for (int attempt = 0;; attempt++) {
123 drainEvents();
124 if (!syscall_mcReadSector(DEVICE, sector, buffer)) return -2;
125 int e = waitCard();
126 if (e != NEWCARD || attempt) return e;
127 syscall_mcAllowNewCard();
128 }
129}

131static int cardWrite(int sector, const uint8_t *buffer) {

134 if (!syscall_mcWriteSector(DEVICE, sector, buffer)) return -2;

137 syscall_mcAllowNewCard();
138 }

161// The same rule as tests/memcard: blank, this suite's own title frame over
162// frame 0, or a Sony directory naming nothing but BIMEMTEST files.
163static int cardIsOursToErase() {
164 static const uint8_t titleSjis[] = {0x82, 0x6c, 0x82, 0x73};
165 int e = cardRead(0, s_frame);
166 if (e != DONE) {
167 ramsyscall_printf("REFUSE: frame 0 read ended on %s\n", e == -2 ? "a refused call" : eventName(e));
168 return 0;
169 }
170 int zeros = 1, ones = 1;
171 for (int i = 0; i < 128; i++) {
172 zeros = zeros && s_frame[i] == 0x00;
173 ones = ones && s_frame[i] == 0xff;
174 }
175 if (zeros || ones) {
176 ramsyscall_printf("card: blank (frame 0 all %s)\n", zeros ? "00" : "ff");
177 return 1;
178 }
179 if (s_frame[0] == 'S' && s_frame[1] == 'C' && same(s_frame + 4, titleSjis, sizeof(titleSjis))) {
180 ramsyscall_printf("card: frame 0 holds the memcard test's title frame, from an earlier run\n");
181 return 1;
182 }
183 if (s_frame[0] != 'M' || s_frame[1] != 'C') {
184 ramsyscall_printf("REFUSE: frame 0 is neither blank nor a Sony header\n");
185 return 0;
186 }
187 for (int s = 1; s <= 15; s++) {
188 e = cardRead(s, s_frame);
189 if (e != DONE) {
190 ramsyscall_printf("REFUSE: directory frame %d read ended on %s\n", s, eventName(e));
191 return 0;
192 }
193 if (s_frame[0] == 0x51 && !startsWith(s_frame + 0x0a, "BIMEMTEST")) {
194 s_frame[0x0a + 20] = 0;
195 ramsyscall_printf("REFUSE: block %d holds a save this test did not write: %s\n", s,
196 (const char *)s_frame + 0x0a);
197 return 0;
198 }
199 }
200 ramsyscall_printf("card: Sony filesystem, no save but this suite's\n");
201 return 1;
202}

204static void testSectors() {
205 int ok = 1;
206 for (int s = 64; s < 67; s++) {
207 fill(s_frame, 128, s);
208 int e = cardWrite(s, s_frame);
209 if (e != DONE) ramsyscall_printf("_card_write %d ended on %s\n", s, eventName(e)), ok = 0;
210 }
211 for (int s = 64; s < 67; s++) {
212 fill(s_frame, 128, s);
213 int e = cardRead(s, s_frame2);
214 if (e != DONE || !same(s_frame, s_frame2, 128)) {
215 ramsyscall_printf("_card_read %d ended on %s%s\n", s, eventName(e), e == DONE ? ", data differs" : "");
216 ok = 0;
217 }
218 }
219 check(ok, "_card_write then _card_read, 3 sectors");
220
221 // The last sector is 1023. The retail kernel's range check lets 1024
222 // through (measured on an SCPH-5501, and OpenBIOS matches it), and the
223 // card is what refuses it, with an error event.
224 drainEvents();
225 int accepted = syscall_mcReadSector(DEVICE, 1024, s_frame2);
226 int e = accepted ? waitCard() : -2;
227 ramsyscall_printf("_card_read 1024: %s, ended on %s\n", accepted ? "accepted" : "refused", eventName(e));
228 check(accepted, "_card_read passes sector 1024 to the card");
229#ifndef PCSX_TESTS
230 // PCSX-Redux's card answers sector 1024 with 128 FF bytes and a good end
231 // byte, so the kernel reports it done.
232 check(e == ERROR, "the card refuses sector 1024");
233#endif
234 drainEvents();
235 accepted = syscall_mcReadSector(DEVICE, 1023, s_frame2);
236 e = accepted ? waitCard() : -2;
237 check(accepted && e == DONE, "_card_read takes sector 1023");
238}

240static void testDevice() {
241 struct DirEntry entry;
242 check(buFormat(BU_PATH("")) == 1, "format");
243 check(firstFile(BU_PATH("*"), &entry) == NULL, "firstfile on a formatted card finds nothing");
244
245 int fd = syscall_open(BU_PATH("BIMEMTEST-BIOS"), PSXF_CREAT | (1 << 16));
246 check(fd >= 0, "open O_CREAT, 1 block");
247 if (fd >= 0) syscall_close(fd);
248
249 fill(s_data, 2048, 1);
250 fd = syscall_open(BU_PATH("BIMEMTEST-BIOS"), PSXF_WRITE);
251 int n = fd >= 0 ? (int)syscall_write(fd, s_data, 1024) : -1;
252 if (fd >= 0) syscall_close(fd);
253 check(n == 1024, "write 1024 bytes");
254
255 fd = syscall_open(BU_PATH("BIMEMTEST-BIOS2"), PSXF_CREAT | (2 << 16));
256 check(fd >= 0, "open O_CREAT, 2 blocks");
257 if (fd >= 0) syscall_close(fd);
258 fill(s_data, 2048, 2);
259 fd = syscall_open(BU_PATH("BIMEMTEST-BIOS2"), PSXF_WRITE);
260 n = fd >= 0 ? (int)syscall_write(fd, s_data, 2048) : -1;
261 if (fd >= 0) syscall_close(fd);
262 check(n == 2048, "write 2048 bytes");
263
264 int found = 0, sizes = 1;
265 for (struct DirEntry *p = firstFile(BU_PATH("*"), &entry); p; p = nextFile(&entry)) {
266 found++;
267 ramsyscall_printf("firstfile/nextfile: %s, %d bytes\n", entry.name, (int)entry.size);
268 if (streq(entry.name, "BIMEMTEST-BIOS"))
269 sizes = sizes && entry.size == 8192;
270 else if (streq(entry.name, "BIMEMTEST-BIOS2"))
271 sizes = sizes && entry.size == 16384;
272 else
273 sizes = 0;
274 if (found > 15) break;
275 }
276 check(found == 2 && sizes, "firstfile/nextfile list both files and their sizes");
277
278 fill(s_data, 2048, 1);
279 __builtin_memset(s_readback, 0, 1024);
280 fd = syscall_open(BU_PATH("BIMEMTEST-BIOS"), PSXF_READ);
281 n = fd >= 0 ? syscall_read(fd, s_readback, 1024) : -1;
282 if (fd >= 0) syscall_close(fd);
283 check(n == 1024 && same(s_data, s_readback, 1024), "read back the first file");
284 fill(s_data, 2048, 2);
285 __builtin_memset(s_readback, 0, 2048);
286 fd = syscall_open(BU_PATH("BIMEMTEST-BIOS2"), PSXF_READ);
287 n = fd >= 0 ? syscall_read(fd, s_readback, 2048) : -1;
288 if (fd >= 0) syscall_close(fd);
289 check(n == 2048 && same(s_data, s_readback, 2048), "read back the second file");
290
291 check(erase(BU_PATH("BIMEMTEST-BIOS")) == 1, "erase the first file");
292 struct DirEntry *p = firstFile(BU_PATH("*"), &entry);
293 check(p && streq(entry.name, "BIMEMTEST-BIOS2") && nextFile(&entry) == NULL, "only the second file is left");
294 check(syscall_open(BU_PATH("BIMEMTEST-BIOS"), PSXF_READ) < 0, "open an erased file fails");
295
296 // What the kernel left in the directory, read raw: block 1 deleted, blocks
297 // 2-3 the second file's chain, in the Sony layout psyqo's filesystem reads.
298 int e1 = cardRead(1, s_frame);
299 int dir1 = e1 == DONE ? s_frame[0] : -1;
300 int e2 = cardRead(2, s_frame);
301 int dir2 = e2 == DONE ? s_frame[0] : -1;
302 int named = e2 == DONE && startsWith(s_frame + 0x0a, "BIMEMTEST-BIOS2");
303 int e3 = cardRead(3, s_frame);
304 int dir3 = e3 == DONE ? s_frame[0] : -1;
305 ramsyscall_printf("directory states: block 1 %02x, block 2 %02x, block 3 %02x\n", dir1, dir2, dir3);
306 check(dir1 == 0xa1 && dir2 == 0x51 && named && dir3 == 0x53, "directory: a1 / 51 named / 53");
307
308 check(erase(BU_PATH("BIMEMTEST-BIOS2")) == 1, "erase the second file");
309 check(firstFile(BU_PATH("*"), &entry) == NULL, "card is empty");
310 check(buFormat(BU_PATH("")) == 1, "cleanup: format");
311}

313int main() {
314 ramsyscall_printf("BIOS memory card calls, port %d (slot %d)\n", MEMCARD_PORT, MEMCARD_PORT + 1);
315 initCard(0);
316 startCard();
317 for (int i = 0; i < EVENTS; i++) {
318 s_events[i] = syscall_openEvent(EVENT_HWCARD, s_specs[i], 0x2000, NULL);
319 syscall_enableEvent(s_events[i]);
320 }
321
322 if (cardIsOursToErase()) {
323 testSectors();
324 testDevice();
325 } else {
326 s_failures++;
327 }
328
329 if (s_failures == 0) {
330 ramsyscall_printf("memcard-bios port %d: all %d checks passed\n", MEMCARD_PORT, s_checks);
331 } else {
332 ramsyscall_printf("memcard-bios port %d: %d of %d checks FAILED\n", MEMCARD_PORT, s_failures, s_checks);
333 }
334 return s_failures == 0 ? 0 : 1;
335}
