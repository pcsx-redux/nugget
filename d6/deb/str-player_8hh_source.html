69 enum class Present : uint8_t {
70 // Show a decoded frame at the first vblank after the NEXT frame has
71 // arrived off the disc, the way Sony's libraries do: the disc sets the
72 // rate, and the vblank wait keeps the swap out of the visible picture.
73 // A 320x240 24-bit frame takes about 480 scanlines to decode, so a
74 // 30 fps movie holds 30 fps at 60 Hz but is shown at 25 fps at 50 Hz,
75 // dropping one frame in six.
76 STREAM,
77 // Switch the display to a frame the moment it is decoded, wherever the
78 // beam is. Keeps 30 fps at 50 Hz, but the swap lands anywhere in the
79 // picture and tears visibly on fast motion.
80 ASAP,
81 };

83 struct Options {
84 bool rgb24 = true;

86 // Setfilter: only play the XA channel matching file/channel. When false,
87 // the drive plays every audio sector it meets.
88 bool xaFilter = false;
89 uint8_t xaFile = 0;
90 uint8_t xaChannel = 0;
91 Present present = Present::STREAM;
92 // Number of frames the reassembly ring holds, and the sector budget of a
93 // single frame. A frame using more chunks than this is dropped.
94 uint8_t frameSlots = 3;
95 uint8_t maxSectorsPerFrame = 12;
96 // Run-level buffer for bsdec, in halfwords; two are allocated. A frame
97 // needing more is dropped. Retail 320x240 movies need up to 26304
98 // (Suikoden II) and 25600 (Castlevania SOTN).
99 uint32_t rlHalfwords = 28672;
100 // Size of the display the frame is centered in, in pixels.
101 uint16_t displayWidth = 320;
102 uint16_t displayHeight = 240;
103 };

123 // VRAM, in buckets of 64 lines, the last one 704 and up.
124 uint32_t decodeLineHistogram[12] = {};
125 };
