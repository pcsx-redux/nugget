143 void resume() {
144 if (!m_handle) return;
145 if (!m_suspended) {
146 m_earlyResume = true;
147 return;
148 }
149 m_suspended = false;
150 m_handle.resume();
151 }
152
162 bool done() {
163 if (!m_handle) return true;
164 bool isDone = m_handle.done();
165 if (isDone) {
166 if constexpr (!std::is_void<T>::value) {
167 m_value = eastl::move(m_handle.promise().m_value);
168 }
169 m_handle.destroy();
170 m_handle = nullptr;
171 }
172 return isDone;
173 }

186 const SafeT &value() const { return m_value; }

257 constexpr bool await_ready() { return handle.done(); }
258
259 void await_suspend(std::coroutine_handle<> h) {
260 handle.promise().m_awaitingCoroutine = h;
261 if (!handle.done()) handle.resume();
262 }
263
264 constexpr T await_resume() {
265 if constexpr (std::is_void<T>::value) {
266 handle.destroy();
267 handle = nullptr;
268 return;
269 } else {
270 auto val = eastl::move(handle.promise().m_value);
271 handle.destroy();
272 handle = nullptr;
273 return val;
274 }
275 }
276 };

278 ChainAwaiter operator co_await() && {
279 auto h = m_handle;
280 m_handle = nullptr;
281 return ChainAwaiter{h};
282 }

418 static_assert(StackSize <= 1024, "The scratchpad is 1024 bytes");

422 public:
