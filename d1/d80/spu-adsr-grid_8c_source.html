2// ADSR step grid: every voice on a slow rate steps on the same samples
3// ==========================================================================
4//
5// Voice 0 keys on, voice 23 keys on 2 to 21 samples later, both with a slow
6// linear attack. Every step voice 23 takes lands on a sample where voice 0
7// also steps (voice 23 is serviced later in the sample, so it shows up to
8// two polls after). An envelope counter started at each key-on would put
9// voice 23's steps the key-on delay after voice 0's instead.

34// Returns how many of voice 23's steps were checked against voice 0's;
35// *worst gets the largest lag from a voice 23 step back to the latest voice 0 step.
36static int adsr_grid_run(uint32_t adsr, uint32_t delay, uint32_t *worst) {
37 uint32_t ta[ADSR_GRID_STEPS_A], tb[ADSR_GRID_STEPS_B];
38 int na = 0, nb = 0, bOn = 0;
39 uint16_t la = 0, lb = 0;
40
41 SPU_VOICES[0].adsrHi = (uint16_t)(RELEASE(0, 0) >> 16);
42 SPU_VOICES[23].adsrHi = (uint16_t)(RELEASE(0, 0) >> 16);
43 SPU_KEY_OFF_LOW = 0xffff; SPU_KEY_OFF_HIGH = 0xffff;
44 spu_busy_wait(20000);
45 adsr_grid_setup(0, adsr);
46 adsr_grid_setup(23, adsr);
47 SPU_KEY_OFF_LOW = 0; SPU_KEY_OFF_HIGH = 0;
48 spu_busy_wait(2000);
49
50 COUNTERS[2].mode = 0;
51 uint16_t last = COUNTERS[2].value;
52 uint32_t t = 0, next = 0;
53 SPU_KEY_ON_LOW = 1;
54 while (t < 2000000 && (na < ADSR_GRID_STEPS_A || nb < ADSR_GRID_STEPS_B)) {
55 uint16_t now = COUNTERS[2].value;
56 t += (uint16_t)(now - last);
57 last = now;
58 if (!bOn && t >= delay) { SPU_KEY_ON_HIGH = 1u << 7; bOn = 1; }
59 if (t >= next) {
60 uint16_t a = SPU_VOICES[0].currentVolume, b = SPU_VOICES[23].currentVolume;
61 if (a != la && na < ADSR_GRID_STEPS_A) { ta[na++] = t; la = a; }
62 if (bOn && b != lb && nb < ADSR_GRID_STEPS_B) { tb[nb++] = t; lb = b; }
63 next += ADSR_GRID_POLL;
64 }
65 }
66 SPU_KEY_OFF_LOW = 0xffff; SPU_KEY_OFF_HIGH = 0xffff;
67
68 int checked = 0;
69 *worst = 0;
70 for (int i = 0; i < nb; i++) {
71 if (na == 0 || tb[i] > ta[na - 1] + ADSR_GRID_MAX_LAG) break;
72 uint32_t lag = 0xffffffffu;
73 for (int j = 0; j < na; j++) {
74 if (ta[j] <= tb[i] && tb[i] - ta[j] < lag) lag = tb[i] - ta[j];
75 }
76 if (lag > *worst) *worst = lag;
77 checked++;
78 }
79 return checked;
80}
