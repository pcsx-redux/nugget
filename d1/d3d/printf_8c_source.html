43// C99 lets vsnprintf take a null buffer when n is 0, and it still has to
44// return the length the full output would have had. EASTL's sprintf measures
45// that way before formatting, so nothing may land at address 0.
46
47CESTER_TEST(snprintfNullMeasures, test_instance,
48 volatile char * zero = nullByte();
49 char saved = *zero;
50 int r;
51
52 *zero = 'Z';
53 r = snprintf(NULL, 0, "%d-%s", 42, "ab");
