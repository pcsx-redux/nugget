103class Model {
104 public:
109 bool parse(const uint8_t* blob) {
110 auto header = reinterpret_cast<const Header*>(blob);
111 if (header->magic[0] != 'P' || header->magic[1] != 'X' || header->magic[2] != 'M' ||
112 header->magic[3] != '2') {
113 return false;
114 }
115 m_header = header;
116 m_vertices = reinterpret_cast<const Vec3*>(blob + header->vertexOffset);
117 m_normals = reinterpret_cast<const Vec3*>(blob + header->normalOffset);
118 m_triangles = reinterpret_cast<const Triangle*>(blob + header->triangleOffset);
119 return true;
120 }
121
122 unsigned vertexCount() const { return m_header->vertexCount; }
123 unsigned normalCount() const { return m_header->normalCount; }
124 unsigned triangleCount() const { return m_header->triangleCount; }
125
126 const Vec3* vertices() const { return m_vertices; }
127 const Vec3* normals() const { return m_normals; }
128 const Triangle* triangles() const { return m_triangles; }
129
130 private:
131 const Header* m_header = nullptr;
132 const Vec3* m_vertices = nullptr;
133 const Vec3* m_normals = nullptr;
134 const Triangle* m_triangles = nullptr;
135};
