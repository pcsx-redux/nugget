27// GP0(0xA0) CPU-to-VRAM upload under the E6 mask setting. Same rules as the
28// GP0(0x80) copy in copy.c: set-mask forces bit 15 on every uploaded pixel,
29// and check-mask leaves destination pixels that already have bit 15 set alone.
30// rasterFillRect is the GP0(0xA0) upload, so the tests below drive it with
31// E6 set, and prepare destinations with E6 clear.
32
33// clang-format off
34
35// --------------------------------------------------------------------------
36// 6. The upload honors set-mask: 0x001F lands as 0x801F, and 0x0000 lands as
37// 0x8000, so an uploaded transparent CLUT entry becomes opaque black.
38// --------------------------------------------------------------------------
39CESTER_TEST(vram_upload_honors_set_mask, gpu_vram_move,
40 rasterReset();
41 rasterClearTestRegion(0, 0, 128, 64);
42 rasterSetMaskCtrl(1, 0); // set-mask on
43 rasterFillRect(16, 8, 8, 8, RASTER_VRAM_RED);
44 rasterFillRect(32, 8, 8, 8, 0x0000);
45 rasterSetMaskCtrl(0, 0);
46 ASSERT_PIXEL_EQ(VM_RED_MASKED, 18, 10);
47 ASSERT_PIXEL_EQ(0x8000, 34, 10);
48 ASSERT_PIXEL_UNTOUCHED(48, 10);
49)
50
51// --------------------------------------------------------------------------
52// 7. The upload honors check-mask: a destination pixel with bit 15 set keeps
53// its value, one with bit 15 clear takes the uploaded pixel.
54// --------------------------------------------------------------------------
55CESTER_TEST(vram_upload_honors_check_mask, gpu_vram_move,
56 rasterReset();
57 rasterClearTestRegion(0, 0, 128, 64);
58 rasterFillRect(16, 8, 8, 8, VM_GREEN_MASKED); // protected
59 rasterFillRect(24, 8, 8, 8, RASTER_VRAM_GREEN); // not protected
60 rasterSetMaskCtrl(0, 1); // check-mask on
61 rasterFillRect(16, 8, 16, 8, RASTER_VRAM_RED);
62 rasterSetMaskCtrl(0, 0);
63 ASSERT_PIXEL_EQ(VM_GREEN_MASKED, 18, 10);
64 ASSERT_PIXEL_EQ(RASTER_VRAM_RED, 26, 10);
65)
