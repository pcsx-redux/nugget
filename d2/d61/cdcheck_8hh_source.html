41// Replaces the original shell's hand-rolled CD-ROM state machine (cdrom.c).
42// The original walked the controller protocol by hand each frame: reset, get
43// TOC/ID, read the PVD, validate the "CD001" signature, read the root
44// directory and scan it for PSX.EXE;1 / SYSTEM.CNF;1. psyqo's CDRomDevice and
45// ISO9660Parser do all of that, so this is a coroutine that loops the same
46// decision and publishes an observable state for the scene to react to.
47//
48// Note: like the original, audio-CD detection is not implemented (the original
49// defined CD_SUCCESS_AUDIO but never reached it, so isCDAudio() was always
50// false). isAudio() is kept as a faithful stub.
51class CDChecker {
52 public:
53 enum class State {
54 Checking, // probe in progress, no verdict yet
55 Error, // no recognizable PlayStation disc (sticky across retries)
56 Success, // a bootable data disc was found
57 };
58
59 // Must be called from the Application's prepare() (after gpu().initialize()).
60 void prepare(psyqo::GPU &gpu);
61 // Kicks off the probe coroutine; call once from createScene().
62 void start();
63
64 State state() const { return m_state; }
65 bool isError() const { return m_state == State::Error; }
66 bool isSuccess() const { return m_state == State::Success; }
67 bool isAudio() const { return false; }
68
69 private:
70 psyqo::Coroutine<> probe();
71
72 psyqo::GPU *m_gpu = nullptr;
73 psyqo::CDRomDevice m_cdrom;
74 psyqo::ISO9660Parser m_isoParser = psyqo::ISO9660Parser(&m_cdrom);
75 psyqo::Coroutine<> m_coroutine;
76 State m_state = State::Checking;
77};
