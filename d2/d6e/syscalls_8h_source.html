39#ifndef PS1_PC_PORT
40// Each BIOS call goes through a two-instruction thunk, `j <table>` with t1 set
41// in its delay slot, emitted in every file that includes this header as a weak
42// symbol in its own COMDAT section, so the linker keeps one copy of the ones
43// that are used. The thunks are plain functions to the compiler: t1 cannot be
44// reused before the jump, the BIOS returns straight to the caller, and a call
45// in tail position stays a jump.
46// Code spread over more than one 256 MB region (OpenBIOS runs from ROM and
47// RAM) defines NUGGET_BIOS_LONG_CALLS, so the thunks are reached through a
48// register rather than a jal. It has to be set for every object in a link or
49// for none: both forms of a thunk share one name and one COMDAT group, so the
50// linker keeps whichever copy it sees first without a diagnostic.
51// Every included thunk is emitted in every object, so a link needs
52// --gc-sections (common.mk passes it) to drop the ones nothing calls.
53#ifdef NUGGET_BIOS_LONG_CALLS
54#define NUGGET_BIOS_CALL_ATTR __attribute__((long_call))
55#else
56#define NUGGET_BIOS_CALL_ATTR
57#endif
58// A j reaches 0xa0 only from code in the same 256 MB region, which ROM code
59// is not, so long-call code jumps through t2 (the BIOS handlers clobber it).
60#ifdef NUGGET_BIOS_LONG_CALLS
61#define NUGGET_BIOS_JUMP(table) " li $t2, " #table "\n jr $t2\n"
62#else
63#define NUGGET_BIOS_JUMP(table) " j " #table "\n"
64#endif
65#ifdef __cplusplus
66#define NUGGET_BIOS_EXTERN extern "C" NUGGET_BIOS_CALL_ATTR
67#else
68#define NUGGET_BIOS_EXTERN extern NUGGET_BIOS_CALL_ATTR
69#endif
70#define NUGGET_BIOS_THUNK(name, table, number) \
71 __asm__(".ifndef __nugget_bios_" #name \
72 "\n" \
73 ".pushsection .text.__nugget_bios_" #name ",\"axG\",@progbits,__nugget_bios_" #name \
74 ",comdat\n" \
75 ".weak __nugget_bios_" #name \
76 "\n" \
77 ".type __nugget_bios_" #name \
78 ", @function\n" \
79 ".set push\n.set noreorder\n" \
80 "__nugget_bios_" #name ":\n" NUGGET_BIOS_JUMP(table) " li $t1, " #number \
81 "\n" \
82 ".set pop\n" \
83 ".size __nugget_bios_" #name \
84 ", . - __nugget_bios_" #name \
85 "\n" \
86 ".popsection\n" \
87 ".endif\n");
88#endif

91static __attribute__((always_inline)) int enterCriticalSection() {
92 register int n asm("a0") = 1;
93 register int r asm("v0");
94 __asm__ volatile("syscall\n" : "=r"(r) : "r"(n) : "memory");
95 return r;
96}
97
98static __attribute__((always_inline)) void leaveCriticalSection() {
99 register int n asm("a0") = 2;
100 __asm__ volatile("syscall\n" : : "r"(n) : "memory");
101}

103static __attribute__((always_inline)) int changeThreadSubFunction(uint32_t address) {
104 register int n asm("a0") = 3;
105 register int tcb asm("a1") = address;
106 register int r asm("v0");
107 __asm__ volatile("syscall\n" : "=r"(r) : "r"(n), "r"(tcb) : "memory");
108 return r;
109}
