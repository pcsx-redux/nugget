31/* Frame checksum: Fletcher-32 over the 16-bit word stream (TYPE, LEN, payload),
32 two register accumulators with the modulo deferred to the end. 0 on the wire
33 means "not computed" and is accepted without verifying; a computed value of
34 exactly 0 is sent as 0xffffffff so the sentinel stays unambiguous. */
35#define CKSUM_NONE 0u
36static uint32_t s_txS1, s_txS2;
37static uint32_t s_rxS1, s_rxS2;
38
39static uint32_t fletcherFinish(uint32_t s1, uint32_t s2) {
40 uint32_t ck = ((s2 % 65535u) << 16) | (s1 % 65535u);
41 return ck == CKSUM_NONE ? 0xffffffffu : ck;
42}

46/* Running checksum for the streaming send API. The monitor sends exactly one
47 frame at a time, so a single static accumulator is safe. */
48
49void transportSendBegin(uint16_t type, uint16_t len) {
50 s_txS1 = s_txS2 = 0;
51#ifdef MONITOR_LINK_IS_STREAM
52 linkPutByte(0); /* leaves console text, a frame follows */
53#endif
54 linkPutWord(FRAME_SYNC); /* SYNC is outside the checksum */
55 linkPutWord(type);
56 s_txS1 += type; s_txS2 += s_txS1;
57 linkPutWord(len);
58 s_txS1 += len; s_txS2 += s_txS1;
59}
60
61void transportSendWord(uint16_t w) {
62 linkPutWord(w);
63 s_txS1 += w; s_txS2 += s_txS1;
64}
65
66void transportSendEnd(void) {
67 uint32_t ck = fletcherFinish(s_txS1, s_txS2);
68 linkPutWord((uint16_t)(ck & 0xffff)); /* low word first */
69 linkPutWord((uint16_t)((ck >> 16) & 0xffff)); /* high word */
70}

72void transportSendFrame(uint16_t type, const uint16_t *payload, uint16_t len) {
73 transportSendBegin(type, len);
74 for (uint16_t i = 0; i < len; i++) transportSendWord(payload[i]);
75 transportSendEnd();
76}

84void transportRecvBegin(uint16_t *type, uint16_t *len) {
85 uint16_t t, l;
86 linkRxOpen();
87 for (;;) {
88 uint8_t b = linkGetByte();
89 if (b != 0) {
90 s_consoleDropped++;
91 continue;
92 }
93 /* A run of 0s is one frame start; a 0 not followed by SYNC, or a LEN
94 no frame can have, is noise. */
95 uint8_t lo;
96 while ((lo = linkGetByte()) == 0) {
97 }
98 if ((lo | (linkGetByte() << 8)) != FRAME_SYNC) continue;
99 t = linkGetWord();
100 l = linkGetWord();
101 if (l <= TRANSPORT_STREAM_MAX_LEN) break;
102 }
103 s_rxS1 = s_rxS2 = 0;
104 s_rxS1 += t; s_rxS2 += s_rxS1;
105 s_rxS1 += l; s_rxS2 += s_rxS1;
106 *type = t;
107 *len = l;
108}

130int transportRecvEnd(void) {
131 uint32_t rxck = linkGetWord();
132 rxck |= ((uint32_t)linkGetWord()) << 16;
133#ifdef MONITOR_LINK_IS_STREAM
134 linkRxClose();
135 if (rxck == CKSUM_NONE) return TRANSPORT_ECKSUM; /* mandatory on a byte link */
136#else
137 if (rxck == CKSUM_NONE) return TRANSPORT_OK; /* sender skipped it */
138#endif
139 return (rxck == fletcherFinish(s_rxS1, s_rxS2)) ? TRANSPORT_OK : TRANSPORT_ECKSUM;
140}

147/* The frames accepted while trying a new rate: PING with no payload, then,
148 once the host has seen the PONG, PING carrying the word 1. The host
149 retries the first freely, so the second has to differ from it. */
150static const uint8_t s_pingFrame[] = {0x00, 0xaa, 0x55, 0x01, 0x00, 0x00, 0x00, 0x01, 0x00, 0x02, 0x00};
151static const uint8_t s_confirmFrame[] = {0x00, 0xaa, 0x55, 0x01, 0x00, 0x01, 0x00, 0x01, 0x00, 0x03, 0x00, 0x06, 0x00};

155/* Wait up to one window for the exact frame `f` at the current rate. */
156static int awaitFrame(const uint8_t *f, unsigned n) {
157 unsigned matched = 0;
158 linkRxOpen();
159 for (uint32_t spins = 0; spins < MONITOR_RATE_WINDOW_SPINS; spins++) {
160 uint8_t b;
161 if (!linkTryGetByte(&b)) continue;
162 if (b == f[matched]) {
163 if (++matched == n) {
164 linkRxClose();
165 return 1;
166 }
167 } else {
168 matched = (b == f[0]) ? 1 : 0;
169 }
170 }
171 linkRxClose();
172 return 0;
173}

175int transportTryRate(uint16_t reload, uint16_t pongType, uint16_t pongWord) {
176 uint16_t old = linkGetRate();
177 linkSetRate(reload);
178 /* The PING proves host -> PS1 at the new rate. Only a host that got the
179 PONG sends the confirmation, which proves PS1 -> host. */
180 if (awaitFrame(s_pingFrame, sizeof(s_pingFrame))) {
181 transportSendFrame(pongType, &pongWord, 1);
182 if (awaitFrame(s_confirmFrame, sizeof(s_confirmFrame))) return 1;
183 }
184 linkSetRate(old);
185 return 0;
186}

192int transportRecvFrame(uint16_t *type, uint16_t *payload, uint16_t maxLen, uint16_t *lenOut) {
193 uint16_t t, len;
194 transportRecvBegin(&t, &len);
195
196 if (len > maxLen) {
197 for (uint16_t i = 0; i < len; i++) transportRecvWord(); /* drain, stay aligned */
198 transportRecvEnd();
199 return TRANSPORT_EBADLEN;
200 }
201
202 for (uint16_t i = 0; i < len; i++) payload[i] = transportRecvWord();
203 int rc = transportRecvEnd();
204 if (rc != TRANSPORT_OK) return rc;
205
206 *type = t;
207 *lenOut = len;
208 return TRANSPORT_OK;
209}
