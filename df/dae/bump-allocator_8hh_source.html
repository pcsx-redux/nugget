55template <size_t N, Safe safety = Safe::Yes>
56class BumpAllocator {
57 public:
58 template <Primitive P, typename... Args>
59 Fragments::SimpleFragment<P> &allocateFragment(Args &&...args) {
60 static constexpr size_t size = sizeof(Fragments::SimpleFragment<P>);
61 if constexpr (safety == Safe::Yes) {
62 psyqo::Kernel::assert(remaining() >= size, "BumpAllocator: Out of memory");
63 }
64 uint8_t *ptr = m_current;
65 m_current += size;
66 return *new (ptr) Fragments::SimpleFragment<P>(eastl::forward<Args>(args)...);
67 }
68 template <typename T, typename... Args>
69 T &allocate(Args &&...args) {
70 // Rounding to 4 keeps the cursor 4-aligned, so fragments allocated
71 // afterwards stay aligned.
72 size_t size = (sizeof(T) + 3) & ~size_t(3);
73 uint8_t *ptr = m_current;
74 if constexpr (alignof(T) > 1) {
75 static constexpr size_t a = alignof(T) - 1;
76 auto alignedptr = reinterpret_cast<uint8_t *>((reinterpret_cast<uintptr_t>(ptr) + a) & ~a);
77 size += alignedptr - ptr;
78 ptr = alignedptr;
79 }
80 if constexpr (safety == Safe::Yes) {
81 psyqo::Kernel::assert(remaining() >= size, "BumpAllocator: Out of memory");
82 }
83 m_current += size;
84 return *new (ptr) T(eastl::forward<Args>(args)...);
85 }
86 void reset() { m_current = m_memory; }
87 size_t remaining() const { return N - (m_current - m_memory); }
88 size_t used() const { return m_current - m_memory; }
89
90 private:
91 uint8_t m_memory[N] __attribute__((aligned(4)));
92 uint8_t *m_current = m_memory;
93};
