122 [[gnu::always_inline, gnu::no_instrument_function]] static inline uint32_t now() {
123 volatile uint16_t* const counter1 = reinterpret_cast<volatile uint16_t*>(0xbf801110);
124 volatile uint16_t* const counter2 = reinterpret_cast<volatile uint16_t*>(0xbf801120);
125 uint32_t lines, cycles;
126 do {
127 lines = *counter1;
128 cycles = *counter2;
129 } while (lines != *counter1);
130 return (lines << 16) | cycles;
131 }

165 struct BenchResult {
166 uint32_t count = 0;
167 uint32_t min = 0;
168 uint32_t max = 0;

170 uint32_t mean() const { return count ? total / count : 0; }

182 template <typename F>
183 static BenchResult bench(const Zone& zone, unsigned count, F&& function) {
184 BenchResult result;
185 if (!s_header || !count) return result;
186 uint32_t generation = s_generation;
187 uint8_t depth = s_depth;
188 unsigned stackDepth = ShadowStack::depth();
189 if (stackDepth > 255) stackDepth = 255;
190 result.min = 0xffffffff;
191 for (unsigned i = 0; i < count; i++) {
192 uint16_t vsync = *const_cast<const volatile uint32_t*>(&psyqoFrameCounter);
193 s_depth = depth + 1;
194 uint32_t start = now();
195 function();
196 uint32_t end = now();
197 s_depth = depth;
198 // Reinstalled from inside the run: the counters were reset under it.
199 if (s_generation != generation) break;
200 record(&zone, start, end, vsync, depth, stackDepth);
201 uint32_t cycles = elapsed(start, end);
202 cycles = cycles > s_nowCost ? cycles - s_nowCost : 0;
203 if (cycles < result.min) result.min = cycles;
204 if (cycles > result.max) result.max = cycles;
205 result.total += cycles;
206 result.count++;
207 }
208 if (!result.count) result.min = 0;
209 return result;
210 }

219 template <typename F>
220 static void forEach(F&& function) {
221 pcsx_ProfHeader* header = s_header;
222 if (!header) return;
223 uint32_t written = header->written;
224 uint32_t mask = header->capacity - 1;
225 uint32_t count = s_full ? header->capacity : written;
226 auto entries = reinterpret_cast<const pcsx_ProfEntry*>(header->entries);
227 for (uint32_t i = written - count; i != written; i++) {
228 const pcsx_ProfEntry& entry = entries[i & mask];
229 function(*reinterpret_cast<const Zone*>(entry.zone), entry.start, elapsed(entry.start, entry.end),
230 entry.vsync, entry.depth, entry.stackDepth);
231 }
232 }
