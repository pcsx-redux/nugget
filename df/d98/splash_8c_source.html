46// The original version of this function (as found in the 573 BIOS) invokes a
47// subroutine repeatedly in order to draw each color bar, rather than using an
48// array and a loop, and has several completely unnecessary checks. This is a
49// much simpler and more straightforward but functionally equivalent
50// implementation.
51void drawSplashScreen() {
52#ifdef OPENBIOS_SHOW_SPLASH_SCREEN
53 struct DisplayModeConfig config = {.hResolution = HR_256,
54 .vResolution = VR_240,
55 .videoMode = VM_NTSC,
56 .colorDepth = CD_15BITS,
57 .videoInterlace = VI_OFF,
58 .hResolutionExtended = HRE_NORMAL};
59
60 GPU_STATUS = 0x00000000; // Reset
61 setDisplayArea(0, 0);
62 setHorizontalRange(0, 256 * 10);
63 setVerticalRange(16, 255);
64 setDisplayMode(&config);
65
66 const int barCount = sizeof(s_colors) / sizeof(union Color);
67 struct FastFill ff = {.c = 0, .x = 0, .y = 0, .w = 256 / barCount, .h = 240};
68
69 for (int i = 0; i < barCount; i++, ff.x += ff.w) {
70 ff.c.packed = s_colors[i].packed;
71 fastFill(&ff);
72 }
73
74 waitGPU();
75 enableDisplay();
76#endif
77}
