64/* Calls OpenBIOS API entry `index` with two arguments. The call itself is
65 in the asm so that nothing the compiler schedules between setting t1 and
66 the jump can reuse t1; it clobbers what a normal call does. */
67static inline uintptr_t openBiosCall(uint32_t index, uintptr_t arg0, uintptr_t arg1) {
68 uintptr_t* a0table = (uintptr_t*)0x200;
69 uintptr_t fn = a0table[11] ^ 1;
70 register uintptr_t a0 asm("a0") = arg0;
71 register uintptr_t a1 asm("a1") = arg1;
72 register uintptr_t t1 asm("t1") = index;
73 register uintptr_t v0 asm("v0");
74 __asm__ volatile("jalr %4\n\tnop\n"
75 : "=r"(v0), "+r"(a0), "+r"(a1), "+r"(t1)
76 : "r"(fn)
77 : "$1", "v1", "a2", "a3", "t0", "t2", "t3", "t4", "t5", "t6", "t7", "t8", "t9", "ra", "hi", "lo",
78 "memory");
79 return v0;
80}

82/* Nonzero if this OpenBIOS was built with the debug monitor in it
83 (MONITOR=1): that monitor owns the exception path and its link from
84 boot, and answers `break 4, 1` with HELLO. */
85static inline int getOpenBiosMonitor() {
86 if (getOpenBiosApiVersion() < 1) return 0;
87 return (int)openBiosCall(0x02, 0, 0);
88}

90/* Puts a call to `fn` (`lui at / ori at, at / jalr at / nop`) into the
91 kernel exception handler's patch slot `slot` (1..4) and flushes the
92 i-cache, only if that slot is still four nops. `fn` runs with only at,
93 v0, v1 and ra saved into the register frame k0 points at, and must
94 return with k0 intact. Returns 0 when installed; nonzero, with nothing
95 written, for a slot out of range or already taken. */
96static inline int installOpenBiosExceptionSlot(int slot, void (*fn)(void)) {
97 if (getOpenBiosApiVersion() < 1) return -1;
98 return (int)openBiosCall(0x03, (uintptr_t)slot, (uintptr_t)fn);
99}

101/* A RAM range (KSEG0 address and size in bytes) that OpenBIOS guarantees
102 it never uses, for resident code loaded under it. Returns 0 and a size
103 of 0 when there is none to be had. */
104static inline void* getOpenBiosCodeCave(uint32_t* size) {
105 if (getOpenBiosApiVersion() < 1) {
106 if (size) *size = 0;
107 return 0;
108 }
109 return (void*)openBiosCall(0x04, (uintptr_t)size, 0);
110}
