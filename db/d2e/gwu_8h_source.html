26/*
27 * Write-to-use delay, per cop2 command and per input register.
28 *
29 * Every cell is one (command, input register) pair of the gte-latency tests,
30 * with that test's scene and canary. A cell function writes the register's
31 * scene value, waits 4 nops, writes a second value, waits N nops, runs the
32 * command:
33 *
34 * mtc2/ctc2 a, reg; nop x4; mtc2/ctc2 b, reg; nop x N; cop2 CMD
35 *
36 * and the command's outputs are compared against two references run through
37 * the same function: a = b = scene value (old), a = b = canary (new).
38 * The written register itself is left out of the comparison.
39 *
40 * Varied: N = 0..3; A = 0..3, the word within a 16-byte cache line the
41 * sequence starts at (A0..A3, cached); and cached vs uncached fetch (U,
42 * at A = 0). Held: interrupts off, nop as the filler and as the
43 * instruction before the second write, mtc2/ctc2 as the writer, the
44 * measured pass is the second of two back to back.
45 *
46 * The last column of each cell is the cached result from SCPH-1000, -1001,
47 * -5501 (two) -7001 and -9002 (see RESULTS.md). A letter that is the same
48 * at every A is checked against it, and the program returns 1 on any
49 * difference; '?' marks a letter that moves with A or between consoles,
50 * which is printed and not checked. The uncached pass is printed only.
51 */

82// A = how many words past a 16-byte cache line boundary the sequence starts.
83#define GWU_FN(W, name, op, reg, N, A) \
84 static void name##_##N##_##A(uint32_t a, uint32_t b) { \
85 __asm__ volatile(".set push\n.set noreorder\n.balign 16\n.rept " #A "\nnop\n.endr\n" W " %0, $" #reg \
86 "\nnop\nnop\nnop\nnop\n" W " %1, $" #reg "\n.rept " #N \
87 "\nnop\n.endr\ncop2 %2\nnop\nnop\n.set pop\n" \
88 : \
89 : "r"(a), "r"(b), "i"(op)); \
90 }
