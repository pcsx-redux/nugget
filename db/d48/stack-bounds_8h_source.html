43// Each level holds a frame of at least 64 bytes. The array is volatile and the
44// result is used after the call, so the recursion cannot become a loop.
45static __attribute__((noinline)) uint32_t recurse(uint32_t depth) {
46 volatile uint32_t frame[16];
47 for (unsigned i = 0; i < 16; i++) frame[i] = depth + i;
48 if (depth == 0) return frame[0];
49 return recurse(depth - 1) + frame[15];
50}
