1// ==========================================================================
2// Characterize SPU IRQs from the voice ADPCM read pointer
3// ==========================================================================
4//
5// Hardware goldens captured on real silicon. Only the deterministic latch-
6// semantics value is pinned: that an ack re-arms immediately
7// (rearm.secondPolls). The mid-block re-fire count (midblock.hits) read 8 on
8// one SCPH-5501 and 2 on another, three runs each, so it is printed. The
9// poll counts (aligned.polls, rearm.firstPolls, midblock.firstPolls) and the
10// aligned capture (firstNonzero, lastNonzero, nonzeroCount, sum) are left
11// UNSET and printed as OBS lines: they are CPU-vs-SPU timing quantities that
12// vary run-to-run on silicon (nonzeroCount read 22 to 493
13// across 17 hardware runs), so pinning them would assert a
14// value neither hardware nor an emulator reproduces.

72static void spu_irq_make_sample(void) {
73 for (int block = 0; block < 8; block++) {
74 uint8_t *p = &s_irqSample[block * 16];
75 p[0] = 0x00;
76 p[1] = 0x00;
77 if (block == 0) p[1] = 0x06; // loop start
78 if (block == 7) p[1] = 0x03; // loop end
79 uint8_t nybble = (uint8_t)((block + 1) & 7);
80 uint8_t packed = (uint8_t)(nybble | (nybble << 4));
81 for (int i = 2; i < 16; i++) p[i] = packed;
82 }
83}

142static void spu_irq_analyze_capture(SpuIrqMeasurement *m) {
143 uint32_t first = 0xffffffffu;
144 uint32_t last = 0;
145 uint32_t count = 0;
146 uint32_t sum = 0;
147
148 spu_read_sync(0x0800, s_capture, 1024);
149 for (uint32_t i = 0; i < 512; i++) {
150 uint16_t v = s_capture[i];
151 if (v != 0) {
152 if (first == 0xffffffffu) first = i;
153 last = i;
154 count++;
155 sum += v;
156 }
157 }
158
159 m->captureFirstNonzero = first;
160 m->captureLastNonzero = last;
161 m->captureNonzeroCount = count;
162 m->captureSum = sum;
163}

165static void spu_irq_measure_once(uint32_t irqByteAddr, SpuIrqMeasurement *m) {
166 spu_irq_prepare();
167 m->irqAddr = irqByteAddr;
168
169 // The suite disables CPU interrupts globally. For this characterization pass
170 // we deliberately poll SPUSTAT.6 instead of taking IRQ9: it measures the SPU
171 // IRQ latch/ack semantics without re-enabling Unirom/vblank/SIO handlers into
172 // the jitter-sensitive SPU tests. A later interrupt-delivery test can layer on
173 // IMASK/IREG once the SPU-side latch behavior is known.
174 spu_wait_status_bit11_flip();
175 spu_irq_arm(irqByteAddr);
176 spu_irq_start_voice1();
177
178 m->polls = spu_irq_poll_until(SPU_IRQ_POLL_TIMEOUT);
179 m->fired = (m->polls != 0);
180 m->statusAtFire = SPU_STATUS;
181 m->currentVolumeAtFire = SPU_VOICES[1].currentVolume;
182 m->ackCleared = spu_irq_ack_by_disable();
183 m->statusAfterAck = SPU_STATUS;
184 spu_irq_analyze_capture(m);
185
186 SPU_KEY_OFF_LOW = 0xffff;
187 SPU_KEY_OFF_HIGH = 0xffff;
188 muteSpu();
189}
