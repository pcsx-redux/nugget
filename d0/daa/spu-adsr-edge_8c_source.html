1427static uint32_t spu_adsr_edge_hash(const uint16_t *envx, unsigned samples) {
1428 uint32_t h = 2166136261u;
1429 for (unsigned i = 0; i < samples; i++) {
1430 h ^= envx[i];
1431 h *= 16777619u;

1433 h *= 16777619u;
1434 }

1436}

1438static uint32_t spu_adsr_edge_minmax(const uint16_t *envx, unsigned samples) {
1439 uint16_t min = envx[0];
1440 uint16_t max = envx[0];
1441 for (unsigned i = 1; i < samples; i++) {
1442 if (envx[i] < min) min = envx[i];
1443 if (envx[i] > max) max = envx[i];
1444 }
1445 return ((uint32_t)min << 16) | max;
1446}

1456static void spu_adsr_edge_capture_custom(const char *name, uint32_t adsr, uint8_t mode, uint8_t samples, uint8_t keyoffAt, uint16_t *envx) {
1457 switch (mode) {
1458 case 0:
1459 spu_adsr_capture(adsr, envx, samples);
1460 break;
1461 case 1:
1462 spu_adsr_capture_with_keyoff(adsr, envx, samples, keyoffAt);
1463 break;
1464 default:
1465 spu_reset_quiet();
1466 SPU_CTRL = 0x8000 | 0x4000;
1467 SPU_VOL_MAIN_LEFT = 0; SPU_VOL_MAIN_RIGHT = 0;
1468 SPU_VOICES[1].sampleRate = 0x1000;
1469 SPU_VOICES[1].sampleStartAddr = SPU_UPLOAD_ADDR >> 3;
1470 SPU_VOICES[1].sampleRepeatAddr = SPU_UPLOAD_ADDR >> 3;
1471 SPU_VOICES[1].volumeLeft = 0;
1472 SPU_VOICES[1].volumeRight = 0;
1473 SPU_VOICES[1].adsrHi = (uint16_t)(ATTACK_RATE(0x30, 0) | DECAY(0) | SUSTAIN(3, 31, 15, 0, 0));
1474 SPU_VOICES[1].adsrLo = (uint16_t)((SUSTAIN(3, 31, 15, 0, 0) | RELEASE_RATE(0x30, 0)) >> 16);
1475 SPU_KEY_OFF_LOW = 0xffff; SPU_KEY_OFF_HIGH = 0xffff;
1476 spu_wait_status_bit11_flip();
1477 envx[0] = SPU_VOICES[1].currentVolume;
1478 SPU_KEY_OFF_LOW = 0; SPU_KEY_OFF_HIGH = 0;
1479 SPU_KEY_ON_LOW = 1u << 1;
1480 envx[1] = SPU_VOICES[1].currentVolume;
1481 for (unsigned i = 2; i < samples; i++) {
1482 spu_wait_status_bit11_flip();
1483 if (name[0] == 'm' && i == 4) SPU_VOICES[1].currentVolume = (name[12] == 'z') ? 0 : 0x7fff;
1484 if (name[0] == 'b' && name[9] == 'r' && i == 4) SPU_KEY_OFF_LOW = 1u << 1;
1485 envx[i] = SPU_VOICES[1].currentVolume;
1486 }
1487 SPU_KEY_OFF_LOW = 0xffff; SPU_KEY_OFF_HIGH = 0xffff;
1488 muteSpu();
1489 break;
1490 }
1491}
