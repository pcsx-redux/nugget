27/*
28
29Shared layout for the generic allocator descriptor. Included by both guest code
30(which builds a descriptor and registers it) and the emulator (which reads one
31out of guest memory at registration time).
32
33The tracker built on top of this never inspects guest memory to learn about a
34block: every size and pointer comes from the intercepted call itself. That is
35what lets it work against an allocator whose internal layout is unknown. The
36only guest read is the one-time read of the descriptor below, which the guest
37handed over explicitly.
38
39Argument indices are register-relative in the MIPS O32 sense: 0..3 are a0..a3,
40and 4 and up are stack arguments at sp + 0x10 + (index - 4) * 4. Use
41PCSX_ALLOCARG_NONE for a role the entry does not have.
42
43*/
44

47#include <stdint.h>

49/* Bare relative form on purpose: this header is included both from guest builds
50 (include root src/mips/) and from the emulator (include root src/), and the
51 "common/hardware/..." spelling used elsewhere in this directory only resolves
52 for the first of those. */
53#include "pcsxcmd.h"

55#define PCSX_ALLOCARG_NONE 0xff

57enum pcsx_AllocEntryKind {
58 /* Terminator, and the value an uninitialized entry takes. */
59 PCSX_ALLOC_KIND_END = 0,
60 /* ptr = f(.., size, ..) -- malloc and friends, including operator new. */
61 PCSX_ALLOC_KIND_ALLOC = 1,
62 /* ptr = f(.., count, size, ..) -- the tracked size is count * size. */
63 PCSX_ALLOC_KIND_CALLOC = 2,
64 /* ptr = f(.., oldptr, size, ..) */
65 PCSX_ALLOC_KIND_REALLOC = 3,
66 /* f(.., ptr, ..) -- no return value is captured. */
67 PCSX_ALLOC_KIND_FREE = 4,
68 /* f(.., base, size, ..) -- resets the arena and sets its bounds. */
69 PCSX_ALLOC_KIND_INIT = 5,
70};
71
72struct pcsx_AllocEntry {
73 /* Address of the function's FIRST instruction. Not any address inside it:
74 the tracker reads $ra and the incoming argument registers at this point,
75 and the prologue is what destroys both. */
76 uint32_t address;
77 uint8_t kind;
78 /* Index of the argument carrying a size. For CALLOC this is the element
79 size; for INIT this is the arena length in bytes. */
80 uint8_t argSize;
81 /* Index of the argument carrying a pointer (FREE, REALLOC) or the arena
82 base (INIT). */
83 uint8_t argPtr;
84 /* Index of the argument carrying an element count. CALLOC only. */
85 uint8_t argCount;
86};
87
88struct pcsx_AllocDescriptor {
89 /* op = PCSX_CMD_REGISTER_ALLOCATOR. Set size to the whole thing including
90 the entry array, or the emulator cannot tell where the entries end. */
91 struct pcsx_CommandHeader header;
92 /* Guest pointer to a NUL-terminated label for this arena, or 0. Read once,
93 at registration. */
94 uint32_t nameAddr;
95 /* Bounds of the arena. Both 0 means "unknown": the tracker will derive a
96 display range from the allocations it observes, and will say so rather
97 than pretending the gaps are free space. An INIT entry overrides these. */
98 uint32_t heapStart;
99 uint32_t heapEnd;
100 uint32_t entryCount;
101 struct pcsx_AllocEntry entries[];
102};

104/* Finding kinds, matching PCSX::AllocTracker::FindingKind. Index into
105 pcsx_AllocReport::findingCounts. */
106enum pcsx_AllocFinding {
107 PCSX_ALLOCFINDING_DOUBLE_OR_UNKNOWN_FREE = 0,
108 PCSX_ALLOCFINDING_INTERIOR_FREE = 1,
109 PCSX_ALLOCFINDING_UNKNOWN_REALLOC = 2,

117struct pcsx_AllocReport {
118 /* op = PCSX_CMD_ALLOC_REPORT */
119 struct pcsx_CommandHeader header;
120 /* Input: which registered arena to report on, in registration order. */
121 uint32_t arenaIndex;
122 /* Output. */
123 volatile uint32_t arenaCount;
124 volatile uint32_t liveBlocks;
125 volatile uint32_t liveBytes;
126 volatile uint32_t peakBytes;
127 volatile uint32_t allocCount;
128 volatile uint32_t freeCount;
129 volatile uint32_t pendingCaptures;
130 volatile uint32_t findingCounts[PCSX_ALLOCFINDING_COUNT];
131};

133static __inline__ void pcsx_allocReportInit(struct pcsx_AllocReport* report, uint32_t arenaIndex) {
134 pcsx_commandInit(&report->header, PCSX_CMD_ALLOC_REPORT, sizeof(struct pcsx_AllocReport));
135 report->arenaIndex = arenaIndex;
136}

138/* Convenience for the common case: fill in the header for a descriptor holding
139 `count` entries. The caller still fills nameAddr/heapStart/heapEnd/entries. */
140static __inline__ void pcsx_allocDescriptorInit(struct pcsx_AllocDescriptor* descriptor, uint32_t count) {
141 pcsx_commandInit(&descriptor->header, PCSX_CMD_REGISTER_ALLOCATOR,
142 sizeof(struct pcsx_AllocDescriptor) + count * sizeof(struct pcsx_AllocEntry));
143 descriptor->entryCount = count;
144}
