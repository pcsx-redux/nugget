85static void spu_pitch_edge_run_fmod(const uint8_t *modSample64,
86 uint16_t modPitch,
87 const uint8_t *carrierSample64,
88 uint16_t carrierPitch,
89 int enablePmon) {
90 const uint32_t modAddr = SPU_UPLOAD_ADDR;
91 const uint32_t carrierAddr = SPU_UPLOAD_ADDR + 0x100;
92 spu_reset_quiet();
93 spu_pitch_edge_copy_sample(modAddr, modSample64);
94 spu_pitch_edge_copy_sample(carrierAddr, carrierSample64);
95 SPU_CTRL = 0x8000 | 0x4000;
96 SPU_VOL_MAIN_LEFT = 0x3fff;
97 SPU_VOL_MAIN_RIGHT = 0x3fff;
98 SPU_PITCH_MOD_LOW = enablePmon ? (1u << 1) : 0;
99 SPU_PITCH_MOD_HIGH = 0;
100
101 SPU_KEY_OFF_LOW = 0xffff;
102 SPU_KEY_OFF_HIGH = 0xffff;
103 spu_busy_wait(800000);
104
105 spu_wait_status_bit11_flip();
106 spu_pitch_edge_setup_voice(0, modAddr, modPitch, 0);
107 spu_pitch_edge_setup_voice(1, carrierAddr, carrierPitch, 0);
108 SPU_KEY_OFF_LOW = 0;
109 SPU_KEY_OFF_HIGH = 0;
110 SPU_KEY_ON_LOW = (1u << 0) | (1u << 1);
111 spu_wait_status_bit11_flip();
112
113 // On silicon the voice is in the capture by the first flip after the key-on.
114 // An emulator can still be mixing samples from before the key-on when the
115 // flip is seen, so an all-zero capture takes one more flip, a few times at
116 // most. A capture that already holds signal is read exactly as before.
117 spu_read_sync(0x0800, s_capture, 1024);
118 for (int retry = 0; retry < 4 && spu_pitch_edge_capture_is_silent(); retry++) {
119 spu_wait_status_bit11_flip();
120 spu_read_sync(0x0800, s_capture, 1024);
121 }
122 SPU_KEY_OFF_LOW = 0xffff;
123 SPU_KEY_OFF_HIGH = 0xffff;
124 SPU_PITCH_MOD_LOW = 0;
125 muteSpu();
126}
