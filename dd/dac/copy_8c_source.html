31// VRAM pixel value with the mask bit (bit 15) set, for the mask tests.
32#define VM_RED_MASKED (RASTER_VRAM_RED | 0x8000u) // 0x801F
33#define VM_GREEN_MASKED (RASTER_VRAM_GREEN | 0x8000u) // 0x83E0

35// Issue a GP0(0x80) VRAM-to-VRAM copy: (w x h) region from (sx, sy) to
36// (dx, dy). Coordinates and sizes are absolute pixels.
37static void vramMove(int16_t sx, int16_t sy, int16_t dx, int16_t dy,
38 int16_t w, int16_t h) {
39 waitGPU();
40 GPU_DATA = 0x80000000u;
41 GPU_DATA = ((uint32_t)(uint16_t)sy << 16) | (uint32_t)(uint16_t)sx;
42 GPU_DATA = ((uint32_t)(uint16_t)dy << 16) | (uint32_t)(uint16_t)dx;
43 GPU_DATA = ((uint32_t)(uint16_t)h << 16) | (uint32_t)(uint16_t)w;
44 waitGPU();
45}

49// --------------------------------------------------------------------------
50// 1. The copy moves real pixel data, not a flat color. A two-color source
51// (green left half, blue right half) must arrive intact at the dest, and
52// the source must be left unchanged.
53// --------------------------------------------------------------------------
54CESTER_TEST(vram_move_copies_pixel_data, gpu_vram_move,
55 rasterReset();
56 rasterClearTestRegion(0, 0, 128, 64);
57 // Source at (16,8), 16x8: columns 16..23 green, 24..31 blue.
58 rasterFillRect(16, 8, 8, 8, RASTER_VRAM_GREEN);
59 rasterFillRect(24, 8, 8, 8, RASTER_VRAM_BLUE);
60 vramMove(16, 8, 16, 40, 16, 8); // copy down to (16,40)
61 rasterFlushPrimitive();
62 ASSERT_PIXEL_EQ(RASTER_VRAM_GREEN, 18, 44); // dest left half
63 ASSERT_PIXEL_EQ(RASTER_VRAM_BLUE, 28, 44); // dest right half
64 ASSERT_PIXEL_EQ(RASTER_VRAM_GREEN, 18, 10); // source unchanged
65 ASSERT_PIXEL_UNTOUCHED(8, 44); // outside the dest rect

83// --------------------------------------------------------------------------
84// 3. Unlike fast-fill, the copy honors set-mask: it forces bit 15 on every
85// pixel it writes. (Compare gpu-fbrect fbrect_ignores_set_mask.)
86// --------------------------------------------------------------------------
87CESTER_TEST(vram_move_honors_set_mask, gpu_vram_move,
88 rasterReset();
89 rasterClearTestRegion(0, 0, 128, 64);
90 rasterFillRect(16, 8, 8, 8, RASTER_VRAM_RED); // source bit15 = 0
91 rasterSetMaskCtrl(1, 0); // set-mask on
92 vramMove(16, 8, 100, 8, 8, 8);
93 rasterFlushPrimitive();
94 // Copy forces bit 15 -> 0x801F, not the source's 0x001F.
95 ASSERT_PIXEL_EQ(VM_RED_MASKED, 102, 10);

98// --------------------------------------------------------------------------
99// 4. Unlike fast-fill, the copy honors check-mask: a destination pixel that
100// already has bit 15 set is protected from being overwritten. (Compare
101// gpu-fbrect fbrect_ignores_check_mask, which overwrote it.)
102// --------------------------------------------------------------------------
103CESTER_TEST(vram_move_honors_check_mask, gpu_vram_move,
104 rasterReset();
105 rasterClearTestRegion(0, 0, 128, 64);
106 rasterFillRect(16, 8, 8, 8, RASTER_VRAM_RED); // source
107 rasterFillRect(100, 8, 8, 8, VM_GREEN_MASKED); // dest: green, bit15 = 1
108 rasterSetMaskCtrl(0, 1); // check-mask on
109 vramMove(16, 8, 100, 8, 8, 8); // copy red over it
110 rasterFlushPrimitive();
111 // Destination is mask-protected, so it keeps its green, not the red copy.
112 ASSERT_PIXEL_EQ(VM_GREEN_MASKED, 102, 10);
