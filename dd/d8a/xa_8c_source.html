29// XA-ADPCM real-time playback, against the XA region of the test disc (see
30// create-test-iso.lua). Filler sectors carry their LBA in their first 3
31// bytes, Form 2 data sectors carry their LBA and 'D'. The tests read sector
32// heads at every data ready and check which slots of each 16-sector group
33// reached the host, and the ADPBUSY bit of the status register.
34//
35// The disc is read at double speed and the handler is slower than the drive,
36// so some data sectors are missed: the tests check which sectors arrive,
37// never how many.

57 static int xaCommand(uint8_t command, const uint8_t* params, unsigned count, uint8_t* response,
58 uint8_t* responseSize) {
59 CDROM_REG0 = 0;
60 for (unsigned i = 0; i < count; i++) CDROM_REG2 = params[i];
61 CDROM_REG1 = command;
62 uint32_t timeout = 2000000;
63 if (!waitCDRomIRQWithTimeout(&timeout)) return -1;
64 uint8_t cause = ackCDRomCause();
65 uint8_t buffer[16];
66 uint8_t size = readResponse(response ? response : buffer);
67 if (responseSize) *responseSize = size;
68 return cause;
69 }

71 static int xaWaitCause(uint8_t expected, uint32_t timeoutUs) {
72 uint8_t response[16];
73 uint32_t timeout = timeoutUs;
74 if (!waitCDRomIRQWithTimeout(&timeout)) return 0;
75 uint8_t cause = ackCDRomCause();
76 readResponse(response);
77 return cause == expected;
78 }

80 static int xaSeekL(uint32_t lba) {
81 uint32_t total = lba + 150;
82 uint8_t msf[3] = {itob(total / 4500), itob((total / 75) % 60), itob(total % 75)};
83 if (xaCommand(CDL_SETLOC, msf, 3, NULL, NULL) != 3) return 0;
84 if (xaCommand(CDL_SEEKL, NULL, 0, NULL, NULL) != 3) return 0;
85 return xaWaitCause(2, 4000000);

88 // Pause, then drain: data ready can still be queued ahead of the
89 // acknowledge, so anything but the complete is skipped, bounded.
90 static void xaStop() {
91 CDROM_REG0 = 0;
92 CDROM_REG1 = CDL_PAUSE;
93 for (unsigned i = 0; i < 64; i++) {
94 uint8_t response[16];
95 uint32_t timeout = 1000000;
96 if (!waitCDRomIRQWithTimeout(&timeout)) return;
97 uint8_t cause = ackCDRomCause();
98 readResponse(response);
99 if (cause == 2) return;
100 }
101 }

103 static void xaReadHead(uint8_t* head, unsigned count) {
104 CDROM_REG0 = 0;
105 CDROM_REG3 = 0x80;
106 for (unsigned t = 0; t < 50000; t++) {
107 if (CDROM_REG0 & 0x40) break;
108 }
109 for (unsigned i = 0; i < count; i++) head[i] = CDROM_REG2;
110 CDROM_REG0 = 0;
111 CDROM_REG3 = 0;
112 }

114 // Sets the mode and the filter, ReadS from startLBA for up to durationUs,
115 // and records what reached the host. stopAfterClear ends the window once
116 // that many events after ADPBUSY cleared have been seen.
117 static int xaRead(uint8_t mode, int filter, uint8_t file, uint8_t channel, uint32_t startLBA,
118 uint32_t durationUs, unsigned stopAfterClear, struct XAReadStats* stats) {
119 __builtin_memset(stats, 0, sizeof(*stats));
120 stats->minLBA = 0xffffffff;
121 if (!resetCDRom()) return 0;
122 if (xaCommand(CDL_SETMODE, &mode, 1, NULL, NULL) != 3) return 0;
123 if (filter) {
124 uint8_t params[2] = {file, channel};
125 if (xaCommand(CDL_SETFILTER, params, 2, NULL, NULL) != 3) return 0;
126 }
127 if (!xaSeekL(startLBA)) return 0;
128 if (xaCommand(CDL_READS, NULL, 0, NULL, NULL) != 3) return 0;
129
130 initializeTime();
131 uint32_t elapsed = 0;
132 while ((elapsed < durationUs) && (stats->events < 2000)) {
133 uint32_t slice = 30000;
134 int got = waitCDRomIRQWithTimeout(&slice);
135 elapsed = slice;
136 if (!got) continue;
137 int busy = (CDROM_REG0 & 0x04) != 0;
138 uint8_t cause = ackCDRomCause();
139 uint8_t response[16];
140 readResponse(response);
141 stats->events++;
142 if (cause != 1) {
143 stats->nonDataCauses++;
144 continue;
145 }
146 // With mode bit 5 the whole sector from its header is read, so the
147 // subheader comes first and the payload starts at byte 12.
148 uint8_t head[16];
149 const uint8_t* payload = head;
150 if (mode & 0x20) {
151 xaReadHead(head, 16);
152 payload = head + 12;
153 uint8_t submode = head[6];
154 if (submode & 0x04) {
155 if (submode & 0x40) {
156 stats->audioRealTime++;
157 } else {
158 stats->audioNotRealTime++;
159 }
160 continue;
161 }
162 } else {
163 xaReadHead(head, 4);
164 }
165 uint32_t lba = payload[0] | (payload[1] << 8) | (payload[2] << 16);
166 // Anything without an LBA stamp is a sector that should not have
167 // reached the host, most likely an ADPCM one.
168 if (lba < startLBA || lba > startLBA + 20000) {
169 stats->unstamped++;
170 continue;
171 }
172 if (lba < stats->minLBA) stats->minLBA = lba;
173 if (lba > stats->maxLBA) stats->maxLBA = lba;
174 stats->slotMask |= 1 << ((lba - startLBA) % 16);
175 if (payload[3] == 'D') stats->form2Seen++;
176 if (busy) {
177 stats->busySet++;
178 stats->lastLBABusy = lba;
179 if (stats->busyClear) stats->busyAfterClear++;
180 } else {
181 if (!stats->busyClear) stats->firstLBAClear = lba;
182 stats->busyClear++;
183 if (stopAfterClear && (stats->busyClear >= stopAfterClear)) break;
184 }
185 }
186 xaStop();
187 ramsyscall_printf(
188 "XA mode %02x filter %i %i/%i from %i: %i events, %i not data ready, %i unstamped, slots %04x, LBA %i-%i, "
189 "form 2 %i, audio RT %i not RT %i, ADPBUSY set %i clear %i (last set %i, first clear %i)\n",
190 mode, filter, file, channel, startLBA, stats->events, stats->nonDataCauses, stats->unstamped, stats->slotMask,
191 stats->minLBA, stats->maxLBA, stats->form2Seen, stats->audioRealTime, stats->audioNotRealTime, stats->busySet, stats->busyClear, stats->lastLBABusy,
192 stats->firstLBAClear);
193 return 1;
194 }
