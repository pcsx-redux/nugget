27// CD-DA level through the ATV matrix, read back from the SPU CD capture
28// buffer. Track 2 of the test disc is a 450 Hz sine, identical on both
29// channels, with a peak of 24987.
30
31// Pre-fill for the capture region, small enough that an untouched buffer fails every level check.
32#define CDDA_MARKER 0x0101
33#define CDDA_SPU_DELAY (*(volatile uint32_t *)0xbf801014)

40 static int cddaSpuDma(uint32_t spuAddr, void *buf, uint32_t bytes, int isRead) {
41 const uint16_t tsa = spuAddr >> 3;
42 const uint16_t mode = isRead ? 0x30 : 0x20;
43 SPU_RAM_DTA = tsa;
44 for (int i = 0; i < 0xf01 && SPU_RAM_DTA != tsa; i++);
45 SPU_CTRL = (SPU_CTRL & ~0x30) | mode;
46 for (volatile int i = 0; i < 60; i++);
47 for (int i = 0; i < 0xf01 && (SPU_STATUS & 0x30) != mode; i++);
48 CDDA_SPU_DELAY = (CDDA_SPU_DELAY & 0xf0ffffff) | (isRead ? 0x22000000 : 0x20000000);
49 DMA_CTRL[DMA_SPU].MADR = (uint32_t)buf & 0x1fffffff;
50 DMA_CTRL[DMA_SPU].BCR = ((bytes >> 6) << 16) | 0x10;
51 DMA_CTRL[DMA_SPU].CHCR = isRead ? 0x01000200 : 0x01000201;
52 int ok = 0;
53 for (unsigned t = 0; t < 4000000u; t++) {
54 if ((DMA_CTRL[DMA_SPU].CHCR & 0x01000000) == 0) {
55 ok = 1;
56 break;
57 }
58 }
59 SPU_CTRL = SPU_CTRL & ~0x30;
60 for (volatile int i = 0; i < 60; i++);
61 return ok;
62 }

64 static int cddaSimpleCmd(uint8_t cmd) {
65 uint8_t response[16];
66 CDROM_REG0 = 0;
67 CDROM_REG1 = cmd;
68 uint32_t timeout = 2000000;
69 int ok = waitCDRomIRQWithTimeout(&timeout);
70 ackCDRomCause();
71 readResponse(response);
72 return ok;
73 }

75 static void cddaWaitPauseComplete() {
76 initializeTime();
77 uint32_t elapsed = 0;
78 while (elapsed < 5000000u) {
79 uint32_t budget = 200000u;
80 int ok = waitCDRomIRQWithTimeout(&budget);
81 elapsed = budget;
82 if (!ok) continue;
83 uint8_t cause = ackCDRomCause();
84 uint8_t response[16];
85 readResponse(response);
86 if (cause == 2) return;
87 }

90 // Resets the drive, pre-fills the SPU capture region, sets mute or demute and the ATV
91 // matrix, and turns on CD audio in the SPU. 0 on a failed step.
92 static int cddaAudioSetup(const uint8_t atv[4], int demute) {
93 if (!resetCDRom()) return 0;
94
95 DPCR |= 0x000b0000;
96 SPU_CTRL = 0;
97 for (volatile int i = 0; i < 20000; i++);
98 SPU_VOL_MAIN_LEFT = 0;
99 SPU_VOL_MAIN_RIGHT = 0;
100 SPU_KEY_OFF_LOW = 0xffff;
101 SPU_KEY_OFF_HIGH = 0xffff;
102 SPU_RAM_DTC = 4;
103 SPU_CTRL = 0x8001;
104 for (volatile int i = 0; i < 20000; i++);
105 for (int i = 0; i < 0x400; i++) s_cddaCapture[i] = CDDA_MARKER;
106 if (!cddaSpuDma(0x000, s_cddaCapture, 0x800, 0)) return 0;
107
108 if (!cddaSimpleCmd(demute ? CDL_DEMUTE : CDL_MUTE)) return 0;
109
110 CDROM_REG0 = 2;
111 CDROM_REG2 = atv[0];
112 CDROM_REG3 = atv[1];
113 CDROM_REG0 = 3;
114 CDROM_REG1 = atv[2];
115 CDROM_REG2 = atv[3];
116 CDROM_REG0 = 3;
117 CDROM_REG3 = 0x20;
118 CDROM_REG0 = 0;
119
120 SPU_VOL_CD_LEFT = 0x7fff;
121 SPU_VOL_CD_RIGHT = 0x7fff;
122 SPU_CTRL = SPU_CTRL | 0x8001;
123 return 1;
124 }

126 // Peak and number of sign changes of the left side of the capture buffer.
127 static int cddaBufferPeak(unsigned *signChanges) {
128 int peak = 0;
129 unsigned changes = 0;
130 int prev = 0;
131 for (int i = 0; i < 0x200; i++) {
132 int v = (int16_t)s_cddaCapture[i];
133 int a = v < 0 ? -v : v;
134 if (a > peak) peak = a;
135 int sign = v >= 0 ? 1 : -1;
136 if (i && sign != prev) changes++;
137 prev = sign;
138 }
139 *signChanges = changes;
140 return peak;
141 }

143 // Reads the capture buffer back while audio plays. On a SCPH-9002 the capture sometimes
144 // reads all zeros for a while after the audio should have started, so an empty capture is
145 // retried a few times, 250 ms apart. -1 on a failed DMA.
146 static int cddaCapturePeak(unsigned *signChanges) {
147 int peak = 0;
148 for (unsigned attempt = 0; attempt < 8; attempt++) {
149 if (attempt) {
150 initializeTime();
151 while (updateTime() < 250000u);
152 }
153 if (!cddaSpuDma(0x000, s_cddaCapture, 0x800, 1)) return -1;
154 peak = cddaBufferPeak(signChanges);
155 if (peak) {
156 if (attempt) ramsyscall_printf("capture was empty %u time(s) before audio\n", attempt);
157 break;
158 }
159 }
160 return peak;
161 }

163 // Plays track 2 with the given ATV, captures 512 samples per side while playing,
164 // and returns the left side's peak and number of sign changes. -1 on a failed step.
165 static int cddaPeak(const uint8_t atv[4], int demute, unsigned *signChanges) {
166 if (!cddaAudioSetup(atv, demute)) return -1;
167 if (!setMode(0)) return -1;
168
169 CDROM_REG0 = 0;
170 CDROM_REG2 = 0x02;
171 if (!cddaSimpleCmd(CDL_PLAY)) return -1;
172
173 initializeTime();
174 while (updateTime() < 1500000u);
175
176 int peak = cddaCapturePeak(signChanges);
177
178 CDROM_REG0 = 0;
179 CDROM_REG1 = CDL_PAUSE;
180 cddaWaitPauseComplete();
181
182 // Leave the head on the data track: a GetLocL with an audio sector as the last one
183 // read answers with an error, and the next test reads the location after a reset.
184 CDROM_REG0 = 0;
185 CDROM_REG2 = 0x00;
186 CDROM_REG2 = 0x02;
187 CDROM_REG2 = 0x16;
188 if (cddaSimpleCmd(CDL_SETLOC) && cddaSimpleCmd(CDL_SEEKL)) cddaWaitPauseComplete();
189
190 return peak;
191 }

232// CD audio into the reverb unit. SPUCNT bit 2 sends CD audio to the reverb input, and the
233// reverb unit writes its input into the work area, so the work area shows whether CD audio
234// reached it without any capture of the analog output. Bit 0 gates the send as well.
235#define CDDA_REVERB_BASE 0xcff8
236#define CDDA_REVERB_SIZE 0x18040u
237#define CDDA_REVERB_CHUNK 0x2000u
238#define CDDA_REVERB_SLACK 0x100u
239
240CESTER_BODY(
241 static uint8_t s_cddaReverbBuf[CDDA_REVERB_CHUNK + CDDA_REVERB_SLACK] __attribute__((aligned(4)));
242
243 // Delay from psx-spx's reverb presets: the input is written to mLSAME/mRSAME every tick.
244 static const uint16_t c_cddaDelay[32] = {
245 0x0001, 0x0001, 0x7fff, 0x7fff, 0x0000, 0x0000, 0x0000, 0x0000,
246 0x0000, 0x0000, 0x1fff, 0x0fff, 0x1005, 0x0005, 0x0000, 0x0000,
247 0x1005, 0x0005, 0x0000, 0x0000, 0x0000, 0x0000, 0x0000, 0x0000,
248 0x0000, 0x0000, 0x1004, 0x1002, 0x0004, 0x0002, 0x8000, 0x8000,
249 };
250
251 // Zeroes the work area, or counts the halfwords of it that are not zero. Reads ask for
252 // CDDA_REVERB_SLACK more bytes than they keep: the end of a DMA read can come back wrong.
253 static int cddaReverbArea(int count) {
254 int nonZero = 0;
255 uint32_t addr = CDDA_REVERB_BASE * 8;
256 uint32_t left = CDDA_REVERB_SIZE;
257 if (!count) __builtin_memset(s_cddaReverbBuf, 0, sizeof(s_cddaReverbBuf));
258 while (left) {
259 uint32_t n = left > CDDA_REVERB_CHUNK ? CDDA_REVERB_CHUNK : left;
260 if (count) {
261 if (!cddaSpuDma(addr, s_cddaReverbBuf, CDDA_REVERB_CHUNK + CDDA_REVERB_SLACK, 1)) return -1;
262 const uint16_t *h = (const uint16_t *)s_cddaReverbBuf;
263 for (uint32_t i = 0; i < n / 2; i++) nonZero += h[i] != 0;
264 } else {
265 if (!cddaSpuDma(addr, s_cddaReverbBuf, n, 0)) return -1;
266 }
267 addr += n;
268 left -= n;
269 }
270 return nonZero;
271 }

273 // Plays track 2 for a second with reverb on and the given SPUCNT CD bits, and returns how
274 // many halfwords of the work area the reverb unit wrote. -1 on a failed step.
275 static int cddaReverbWrites(uint16_t cdBits) {
276 static const uint8_t atv[4] = {0x80, 0x00, 0x80, 0x00};
277 if (!cddaAudioSetup(atv, 1)) return -1;
278 if (cddaReverbArea(0) < 0) return -1;
279 volatile uint16_t *regs = (volatile uint16_t *)SPU_REVERB;
280 for (int i = 0; i < 32; i++) regs[i] = c_cddaDelay[i];
281 SPU_REVERB_ADDR = CDDA_REVERB_BASE;
282 if (!setMode(0)) return -1;
283
284 CDROM_REG0 = 0;
285 CDROM_REG2 = 0x02;
286 if (!cddaSimpleCmd(CDL_PLAY)) return -1;
287
288 initializeTime();
289 while (updateTime() < 1500000u);
290 SPU_CTRL = 0x8000 | 0x0080 | cdBits;
291 initializeTime();
292 while (updateTime() < 1000000u);
293 SPU_CTRL = 0x8001;
294 for (volatile int i = 0; i < 20000; i++);
295
296 int written = cddaReverbArea(1);
297 SPU_REVERB_ADDR = 0xffff;
298
299 CDROM_REG0 = 0;
300 CDROM_REG1 = CDL_PAUSE;
301 cddaWaitPauseComplete();
302
303 // Same head parking as cddaPeak.
304 CDROM_REG0 = 0;
305 CDROM_REG2 = 0x00;
306 CDROM_REG2 = 0x02;
307 CDROM_REG2 = 0x16;
308 if (cddaSimpleCmd(CDL_SETLOC) && cddaSimpleCmd(CDL_SEEKL)) cddaWaitPauseComplete();
309
310 return written;
311 }
