31/* LZ4 block decoder fed one input byte at a time. Output goes straight to its
32 destination and earlier output is the match history, so consecutive blocks
33 written to one buffer can reference each other (up to 64 KiB back). A match
34 is copied as its length bytes arrive, so no single input byte produces more
35 than 255 bytes of output. */
36
37struct Lz4Stream {
38 uint8_t *out;
39 uint8_t *base;
40 uint32_t count;
41 uint16_t offset;
42 uint8_t token;
43 uint8_t state;
44};
45
46enum {
47 LZ4S_OK = 0,
48 LZ4S_EBADOFFSET = -1,
49 LZ4S_ETRUNCATED = -2,
50};
51
52/* LZ4STREAM_TRUSTED drops the checks for a malformed stream (a match reaching
53 before the start, a zero offset, a block ending mid-sequence), for when the
54 encoder is known good; feed and end then always return LZ4S_OK. */
55
56/* Start a stream writing at dest. Matches may not reach before dest. */
57void lz4StreamInit(struct Lz4Stream *s, void *dest);
58/* Feed one byte of the current block. */
59int lz4StreamFeed(struct Lz4Stream *s, uint8_t b);
60/* End the current block; the next byte fed starts a new one. Fails if the
61 block stopped in the middle of a sequence. */
62int lz4StreamEndBlock(struct Lz4Stream *s);
