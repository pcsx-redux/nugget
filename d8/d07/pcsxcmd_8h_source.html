27/*
28
29Structured command port, on the single address 0x1f8020a4.
30
31WHY ONE ADDRESS. The EXP2 register space is not ours. Martin Korth's tooling
32and the PS2 backwards-compatibility mechanism have both claimed registers in
33there, and no registry says which. It is a minefield rather than a shortage, so
34the right number of steps to take for all future structured features is one.
35
36WHY THE ANSWER COMES BACK IN MEMORY, NOT IN A REGISTER. The msan helpers in
37pcsxhw.h pass arguments in registers, using a hand-written load/store in an asm
38block with the argument registers pinned. That is deliberate and it is the
39point: the asm block forces the registers rather than hoping the allocator put
40them there, so the emulator can read them out of the register file. It is a
41sound technique for a small fixed-arity call.
42
43It is the wrong shape to generalize, though. Every new operation with a new
44argument list needs another hand-pinned asm block, the arity is capped by the
45argument registers, an operation cannot grow a field later without a new entry
46point, and nothing in it can carry a result back except another pinned
47register. A structured, extensible, versioned protocol wants a struct. So here
48the only channel is memory: results live in a volatile zone inside the command
49struct, ordinary C ordering rules apply, and the port itself is a plain
50volatile store with no register participation at all.
51
52WHY IT IS READABLE AT ALL. A write to an unclaimed address is
53INDISTINGUISHABLE FROM SUCCESS on the guest side: PCSX::HW::write32's default
54case stores into the shadow register file and logs "*Unknown 32bit write". So a
55program built against a newer feature, run on an older emulator, writes into a
56hole and is told nothing. pcsx_present() answers "is this Redux", never "does
57this Redux know command N". The fix needs no return channel: the guest poisons
58results.status before issuing the command, and an emulator that does not know
59the port never overwrites it. Absence of a write is the answer.
60
61The older single-purpose registers (putc, the msan family, the PSYQo heap
62metadata) are NOT migrated and stay valid forever. Breaking them would break
63shipped code, and a command struct is the wrong shape for something like putc
64where building the struct would cost more than the operation.
65
66*/

87enum pcsx_Command {
88 /* Reports protocol version and which operations this build implements.
89 Payload is struct pcsx_QueryResult. Always implemented. */
90 PCSX_CMD_QUERY = 0,
91 /* Registers a generic allocator descriptor. Payload is
92 struct pcsx_AllocDescriptor (allocdesc.h). May be issued repeatedly to
93 track several arenas. */
94 PCSX_CMD_REGISTER_ALLOCATOR = 1,
95 /* Reads back what the tracker believes about one arena. Payload is
96 struct pcsx_AllocReport (allocdesc.h). Exists so a guest can check the
97 tracker against its own ground truth, which is the only way to find out
98 whether the tracker is right rather than merely plausible. */
99 PCSX_CMD_ALLOC_REPORT = 2,
100
101 PCSX_CMD_COUNT
102};

104enum pcsx_CommandStatus {
105 PCSX_CMDSTATUS_OK = 0,
106 /* Poison value. The guest writes this before issuing a command; it
107 surviving means nothing answered, which is what running on an emulator
108 without this port looks like. Never written by the emulator. */
109 PCSX_CMDSTATUS_NO_ANSWER = 1,
110 /* The header magic did not match: a stale or uninitialized struct. */
111 PCSX_CMDSTATUS_BAD_MAGIC = 2,
112 /* Understood the port, does not implement that operation. */
113 PCSX_CMDSTATUS_UNKNOWN_OP = 3,
114 /* Understood the operation, the payload was not acceptable. */
115 PCSX_CMDSTATUS_BAD_ARGUMENT = 4,
116 /* The struct, or something it points at, is not in mapped guest memory. */
117 PCSX_CMDSTATUS_UNMAPPED = 5,
118 /* The header's declared size is too small for the operation's payload. */
119 PCSX_CMDSTATUS_SHORT_STRUCT = 6,
120};

122/* Everything the emulator writes back lives here, and only here. Volatile
123 because it changes outside the C abstract machine's view, at the volatile
124 store to the port. Kept as its own zone rather than qualifying the whole
125 struct so that filling in the (plain) input fields still generates ordinary
126 stores. */
127struct pcsx_CommandResults {
128 volatile uint32_t status;
129 /* Operation-specific outputs follow in each payload struct, and are
130 likewise volatile. */
131};

133struct pcsx_CommandHeader {
134 uint32_t magic;
135 uint32_t op;
136 /* Total size of the whole command struct in bytes, header included. This is
137 how a payload grows later without burning a new op code: the emulator
138 reads and writes only the fields the declared size covers. */
139 uint32_t size;
140 struct pcsx_CommandResults results;
141};

143struct pcsx_QueryResult {
144 struct pcsx_CommandHeader header;
145 volatile uint32_t protocolVersion;
146 /* Operations 0 .. opCount-1 exist in this build's enum; the bitmap says
147 which are implemented. */
148 volatile uint32_t opCount;
149 /* Bit N set means operation N is implemented. 128 operations. */
150 volatile uint32_t opBitmap[4];
151};
152
153/* Fill in a header and poison the result. Call before pcsx_command. */
154PCSX_INLINE void pcsx_commandInit(struct pcsx_CommandHeader* header, uint32_t op, uint32_t size) {
155 header->magic = PCSX_CMD_MAGIC;
156 header->op = op;
157 header->size = size;
158 header->results.status = PCSX_CMDSTATUS_NO_ANSWER;
159}
160
161/* Issue a command. The answer is in cmd->results and in any operation-specific
162 volatile fields; nothing is returned here. The memory clobber keeps the
163 input-field stores from sinking past the port write, since the volatile
164 store alone only orders itself against other volatile accesses. */
165PCSX_INLINE void pcsx_command(void* cmd) {
166 PCSX_COMPILER_BARRIER();
167 *((void* volatile* const)PCSX_CMD_PORT) = cmd;
168 PCSX_COMPILER_BARRIER();
169}
170
171/* Non-zero when this emulator implements the given operation. */
172PCSX_INLINE int pcsx_commandSupported(uint32_t op) {
173 struct pcsx_QueryResult query;
174 if (op >= 128) return 0;
175 pcsx_commandInit(&query.header, PCSX_CMD_QUERY, sizeof(query));
176 query.protocolVersion = 0;
177 pcsx_command(&query);
178 if (query.header.results.status != PCSX_CMDSTATUS_OK) return 0;
179 if (query.protocolVersion == 0) return 0;
180 return (query.opBitmap[op >> 5] >> (op & 31)) & 1;
181}
