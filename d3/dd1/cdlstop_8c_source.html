30 // Sends a command with no parameters and records its two responses.
31 // Returns 0 if either response does not arrive within its timeout.
32 static int stopCommand(uint8_t command, uint8_t* cause1, uint8_t* stat1, uint32_t* ackTime, uint8_t* cause2,
33 uint8_t* stat2, uint32_t* completeTime) {
34 uint8_t response[16];
35 *cause1 = *stat1 = *cause2 = *stat2 = 0xff;
36 *ackTime = *completeTime = 0;
37 initializeTime();
38 CDROM_REG0 = 0;
39 CDROM_REG1 = command;
40 uint32_t timeout = 1000000;
41 if (!waitCDRomIRQWithTimeout(&timeout)) return 0;
42 *ackTime = timeout;
43 *cause1 = ackCDRomCause();
44 readResponse(response);
45 *stat1 = response[0];
46 timeout = 10000000;
47 if (!waitCDRomIRQWithTimeout(&timeout)) return 0;
48 *completeTime = timeout;
49 *cause2 = ackCDRomCause();
50 readResponse(response);
51 *stat2 = response[0];
52 return 1;
53 }
