34 // Waits for the next cause within timeout us of the current time base,
35 // acknowledges it and returns it (0 on timeout). The time lands in *time,
36 // the first response byte in *stat.
37 static uint8_t timingNextCause(uint32_t timeout, uint32_t* time, uint8_t* stat) {
38 uint8_t response[16];
39 uint32_t t = timeout;
40 *stat = 0xff;
41 if (!waitCDRomIRQWithTimeout(&t)) return 0;
42 *time = t;
43 uint8_t cause = ackCDRomCause();
44 if (readResponse(response)) *stat = response[0];
45 return cause;
46 }
47
48 // Same, skipping data ready causes, at most 16 of them, so a response
49 // that never comes cannot keep the wait going while sectors arrive.
50 static uint8_t timingNextNonData(uint32_t timeout, uint32_t* time, uint8_t* stat) {
51 uint8_t cause;
52 unsigned skipped = 0;
53 while ((cause = timingNextCause(timeout, time, stat)) == 1) {
54 if (++skipped > 16) return 1;
55 }
56 return cause;
57 }

59 // ReadN at 00:02:16 in the given mode, n sectors acknowledged without
60 // reading them, then Pause. Returns the Pause INT2 time from its INT3,
61 // or 0 if a step failed. Late INT1s around the Pause are skipped.
62 static uint32_t timingPauseAfter(uint8_t mode, unsigned n, uint8_t* readStat, uint8_t* pauseStat) {
63 uint32_t t;
64 uint8_t stat, cause;
65 if (!resetCDRom()) return 0;
66 if (!setMode(mode)) return 0;
67 if (!setLoc(0, 2, 0x16)) return 0;
68 initializeTime();
69 CDROM_REG0 = 0;
70 CDROM_REG1 = CDL_READN;
71 if (timingNextCause(100000, &t, &stat) != 3) return 0;
72 for (unsigned i = 0; i < n; i++) {
73 if (timingNextCause(3000000, &t, readStat) != 1) return 0;
74 }
75 initializeTime();
76 CDROM_REG0 = 0;
77 CDROM_REG1 = CDL_PAUSE;
78 uint32_t ackTime;
79 cause = timingNextNonData(100000, &ackTime, &stat);
80 if (cause != 3) return 0;
81 uint32_t completeTime;
82 cause = timingNextNonData(3000000, &completeTime, pauseStat);
83 if (cause != 2) return 0;
84 return completeTime - ackTime;
85 }
