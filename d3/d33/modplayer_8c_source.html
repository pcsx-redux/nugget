73struct SpuInstrumentData {
74 uint16_t baseAddress;
75 uint8_t finetune;
76 uint8_t volume;
77 // The fields below are only used to locate 9xx sample offsets in the ADPCM data.
78 uint16_t length; // ADPCM bytes
79 uint32_t loopStart; // MOD sample bytes
80 uint32_t loopLength; // MOD sample bytes, 0 for one-shot samples
81 uint8_t skip; // MOD sample bytes dropped at the start of the sample
82 uint8_t pad; // silent samples inserted at the start of the ADPCM data

87static uint16_t readBE16(const void* ptr) {
88 const uint8_t* bytes = (const uint8_t*)ptr;
89 return (bytes[0] << 8) | bytes[1];
90}

92// SPU address of an instrument, started at samplePos, in MOD sample bytes. The SPU can only start
93// on a 16 bytes ADPCM block of 28 samples, so the position is rounded down to its block. This follows
94// the layout modconv uses: the first word of the MOD sample is skipped, unless a loop starts at 0, and
95// looped samples are front-padded with silence so that the loop starts on a block boundary. Both are
96// derived from the instrument's loop fields, which modconv keeps from the MOD file. Files converted by
97// older versions of modconv have no padding, and offsets into their looped samples can land a block late.
98static uint32_t SPUInstrumentAddress(unsigned sampleID, uint32_t samplePos) {
99 const struct SpuInstrumentData* instrument = &s_spuInstrumentData[sampleID];
100 uint32_t address = instrument->baseAddress << 4;
101 if (samplePos == 0) return address;
102 if (instrument->loopLength != 0) {
103 uint32_t loopEnd = instrument->loopStart + instrument->loopLength;
104 // Past the loop end, MOD players are back into the loop.
105 if (samplePos >= loopEnd) {
106 samplePos = instrument->loopStart + (samplePos - instrument->loopStart) % instrument->loopLength;
107 }
108 }
109 if (samplePos < instrument->skip) samplePos = instrument->skip;
110 uint32_t offset = (instrument->pad + samplePos - instrument->skip) / 28 * 16;
111 if ((offset + 16) >= instrument->length) {
112 // Past the end of a one-shot sample: its last block is the silent loop block.
113 offset = instrument->length >= 16 ? instrument->length - 16 : 0;
114 }
115 return address + offset;
116}

152static void SPUUploadInstruments(uint32_t SpuAddr, const uint8_t* data, uint32_t size) {
153 uint32_t bcr = size >> 6;
154 if (size & 0x3f) bcr++;
155 bcr <<= 16;
156 bcr |= 0x10;
157
158 SPU_RAM_DTA = SpuAddr >> 3;
159 SPU_CTRL = (SPU_CTRL & ~0x0030) | 0x0020;
160 while ((SPU_CTRL & 0x0030) != 0x0020);
161 // original code erroneously was doing SBUS_DEV4_CTRL = SBUS_DEV4_CTRL;
162 SBUS_DEV4_CTRL &= ~0x0f000000;
163 DMA_CTRL[DMA_SPU].MADR = (uint32_t)data;
164 DMA_CTRL[DMA_SPU].BCR = bcr;
165 DMA_CTRL[DMA_SPU].CHCR = 0x01000201;
166
167 while ((DMA_CTRL[DMA_SPU].CHCR & 0x01000000) != 0);
168}

209unsigned MOD_Check(const struct MODFileFormat* module) {
210 const char* s = module->signature;
211 if ((s[0] == 'H') && (s[1] == 'I') && (s[2] == 'T')) {
212 return module->signature[3] - '0';
213 } else if ((s[0] == 'H') && (s[1] == 'M')) {
214 return ((module->signature[2] - '0') * 10) + module->signature[3] - '0';
215 }
216 return 0;
217}

246// This function is now more of a helper to calculate the number of hsync
247// values to wait until the next call to MOD_Poll. If the user wants to use
248// another method, they will have to inspect MOD_BPM manually and make their
249// own math based on their own timer.
250static void MOD_SetBPM(unsigned bpm) {
251 MOD_BPM = bpm;
252 // The original code only uses 39000 here but the reality is a bit more
253 // complex than that, as not all clocks are exactly the same, depending
254 // on the machine's region, and the video mode selected.
255
256 uint32_t status = GPU_STATUS;
257 int isPalConsole = *((const char*)0xbfc7ff52) == 'E';
258 int isPal = (status & 0x00100000) != 0;
259 uint32_t base;
260 if (isPal && isPalConsole) { // PAL video on PAL console
261 base = 39062; // 312.5 * 125 * 50.000 / 50 or 314 * 125 * 49.761 / 50
262 } else if (isPal && !isPalConsole) { // PAL video on NTSC console
263 base = 39422; // 312.5 * 125 * 50.460 / 50 or 314 * 125 * 50.219 / 50
264 } else if (!isPal && isPalConsole) { // NTSC video on PAL console
265 base = 38977; // 262.5 * 125 * 59.393 / 50 or 263 * 125 * 59.280 / 50
266 } else { // NTSC video on NTSC console
267 base = 39336; // 262.5 * 125 * 59.940 / 50 or 263 * 125 * 59.826 / 50
268 }
269 MOD_hblanks = base / bpm;
270}

274static uint32_t loadInternal(const struct MODFileFormat* module, const uint8_t* sampleData) {
275 SPUInit();
276 MOD_Channels = MOD_Check(module);
277
278 if (MOD_Channels == 0) return 0;
279
280 uint32_t currentSpuAddress = 0x1010;
281 for (unsigned i = 0; i < 31; i++) {
282 s_spuInstrumentData[i].baseAddress = currentSpuAddress >> 4;
283 s_spuInstrumentData[i].finetune = module->samples[i].finetune;
284 s_spuInstrumentData[i].volume = module->samples[i].volume;
285 uint16_t length = module->samples[i].lenarr[0] * 0x100 + module->samples[i].lenarr[1];
286 // These match the sample layout of modconv, see SPUInstrumentAddress.
287 uint32_t loopStart = readBE16(&module->samples[i].repeatLocation) * 2;
288 uint32_t loopLength = readBE16(&module->samples[i].repeatLength) * 2;
289 if (loopLength <= 2) loopLength = 0;
290 uint8_t skip = ((loopLength != 0) && (loopStart == 0)) ? 0 : 2;
291 s_spuInstrumentData[i].length = length;
292 s_spuInstrumentData[i].loopStart = loopStart;
293 s_spuInstrumentData[i].loopLength = loopLength;
294 s_spuInstrumentData[i].skip = skip;
295 s_spuInstrumentData[i].pad = loopLength != 0 ? (28 - (loopStart - skip) % 28) % 28 : 0;
296 currentSpuAddress += length;
297 }
298
299 MOD_SongLength = module->songLength;
300
301 unsigned maxPatternID = 0;
302 for (unsigned i = 0; i < 128; i++) {
303 if (maxPatternID < module->patternTable[i]) maxPatternID = module->patternTable[i];
304 }
305
306 MOD_ModuleData = (const uint8_t*)&module->patternTable[0];
307
308 if (sampleData && (sampleData != (const uint8_t*)-1)) {
309 SPUUploadInstruments(0x1010, sampleData, currentSpuAddress - 0x1010);
310 } else if (sampleData == NULL) {
311 SPUUploadInstruments(0x1010, MOD_ModuleData + 4 + 128 + MOD_Channels * 0x100 * (maxPatternID + 1),
312 currentSpuAddress - 0x1010);
313 }
314
315 MOD_CurrentOrder = 0;
316 MOD_CurrentPattern = module->patternTable[0];
317 MOD_CurrentRow = 0;
318 MOD_Speed = 6;
319 MOD_Tick = 6;
320 MOD_RowPointer = MOD_ModuleData + 4 + 128 + MOD_CurrentPattern * MOD_Channels * 0x100;
321 // original code goes only up to MOD_Channels; let's reset all 24
322 for (unsigned i = 0; i < 24; i++) SPUResetVoice(i);
323 MOD_ChangeRowNextTick = 0;
324 MOD_ChangeOrderNextTick = 0;
325 MOD_LoopStart = 0;
326 MOD_LoopCount = 0;
327
328 // these two are erroneously missing from the original code, at
329 // least for being able to play more than one music
330 MOD_PatternDelay = 0;
331 __builtin_memset(s_channelData, 0, sizeof(s_channelData));
332
333 SPUUnMute();
334
335 // this one is also missing, and is necessary, for being able to call MOD_Load
336 // after another song that changed the tempo previously
337 MOD_SetBPM(125);
338
339 // the original code would do:
340 // return MOD_Channels;
341 // but we are returning the size for the MOD_Relocate call
342 return 4 + 128 + MOD_Channels * 0x100 * (maxPatternID + 1);
343}

359void MOD_Relocate(uint8_t* s1) {
360 if (MOD_ModuleData == s1) return;
361 unsigned maxPatternID = 0;
362 for (unsigned i = 0; i < 128; i++) {
363 if (maxPatternID < MOD_ModuleData[i]) maxPatternID = MOD_ModuleData[i];
364 }
365
366 size_t n = 4 + 128 + MOD_Channels * 0x100 * (maxPatternID + 1);
367
368 const uint8_t* s2 = MOD_ModuleData;
369 size_t i;
370
371 if (s1 < s2) {
372 for (i = 0; i < n; i++) *s1++ = *s2++;
373 } else if (s1 > s2) {
374 s1 += n;
375 s2 += n;
376 for (i = 0; i < n; i++) *--s1 = *--s2;
377 }
378
379 MOD_ModuleData = s1;
380}

452static void MOD_UpdateEffect() {
453 const uint8_t* rowPointer = MOD_RowPointer;
454 const unsigned channels = MOD_Channels;
455 for (unsigned channel = 0; channel < channels; channel++) {
456 uint8_t effectNibble23 = rowPointer[3];
457 uint8_t effectNibble1 = rowPointer[2] & 0x0f;
458 uint8_t effectNibble2 = effectNibble23 & 0x0f;
459 uint8_t effectNibble3 = effectNibble23 >> 4;
460
461 uint8_t arpeggioTick;
462 int32_t newPeriod;
463 int16_t volume;
464 uint16_t slideTo;
465 uint8_t fx;
466 uint32_t mutation;
467 int8_t newValue;
468
469 struct SPUChannelData* const channelData = &s_channelData[channel];
470
471 switch (effectNibble1) {
472 case 0: // arpeggio
473 if (effectNibble23 == 0) break;
474 arpeggioTick = MOD_Tick;
475 arpeggioTick %= 3;
476 switch (arpeggioTick) {
477 case 0:
478 newPeriod = channelData->period;
479 break;
480 case 1:
481 newPeriod = MOD_PeriodTable[channelData->note + effectNibble3];
482 break;
483 case 2:
484 newPeriod = MOD_PeriodTable[channelData->note + effectNibble2];
485 break;
486 }
487 SETVOICESAMPLERATE(channel, newPeriod);
488 break;
489 case 1: // portamento up
490 newPeriod = channelData->period;
491 newPeriod -= effectNibble23;
492 if (newPeriod < 108) newPeriod = 108;
493 channelData->period = newPeriod;
494 SETVOICESAMPLERATE(channel, newPeriod);
495 break;
496 case 2: // portamento down
497 newPeriod = channelData->period;
498 newPeriod += effectNibble23;
499 if (newPeriod > 907) newPeriod = 907;
500 channelData->period = newPeriod;
501 SETVOICESAMPLERATE(channel, newPeriod);
502 break;
503 case 5:
504 volume = channelData->volume;
505 if (effectNibble23 < 0x10) {
506 volume -= effectNibble23;
507 if (volume < 0) volume = 0;
508 } else {
509 volume += effectNibble3;
510 if (volume > 63) volume = 63;
511 }
512 channelData->volume = volume;
513 SETVOICEVOLUME(channel, volume);
514 /* fall through */
515 case 3: // glissando
516 newPeriod = channelData->period;
517 slideTo = channelData->slideTo;
518 if (newPeriod < slideTo) {
519 newPeriod += channelData->slideSpeed;
520 if (newPeriod > slideTo) newPeriod = slideTo;
521 } else if (newPeriod > slideTo) {
522 newPeriod -= channelData->slideSpeed;
523 if (newPeriod < slideTo) newPeriod = slideTo;
524 }
525 channelData->period = newPeriod;
526 SETVOICESAMPLERATE(channel, newPeriod);
527 break;
528 case 6:
529 volume = channelData->volume;
530 if (effectNibble23 < 0x10) {
531 volume -= effectNibble23;
532 if (volume < 0) volume = 0;
533 } else {
534 volume += effectNibble3;
535 if (volume > 63) volume = 63;
536 }
537 channelData->volume = volume;
538 SETVOICEVOLUME(channel, volume);
539 /* fall through */
540 case 4: // vibrato
541 mutation = channelData->vibrato & 0x1f;
542 switch (channelData->fx[3] & 3) {
543 case 0:
544 case 3: // 3 is technically random
545 mutation = MOD_SineTable[mutation];
546 break;
547 case 1:
548 if (channelData->vibrato < 0) {
549 mutation *= -8;
550 mutation += 0xff;
551 } else {
552 mutation *= 8;
553 }
554 break;
555 case 2:
556 mutation = 0xff;
557 break;
558 }
559 mutation *= channelData->fx[1] >> 4;
560 mutation >>= 7;
561 newPeriod = channelData->period;
562 if (channelData->vibrato < 0) {
563 newPeriod -= mutation;
564 } else {
565 newPeriod += mutation;
566 }
567 newValue = channelData->vibrato;
568 newValue += channelData->fx[1] & 0x0f;
569 if (newValue >= 32) newValue -= 64;
570 channelData->vibrato = newValue;
571 SETVOICESAMPLERATE(channel, newPeriod);
572 break;
573 case 7: // tremolo
574 // fx[0] is the tremolo position, fx[2] holds speed (low) and depth (high),
575 // and the tremolo waveform is the high nibble of fx[3] (E7x).
576 mutation = channelData->fx[0] & 0x1f;
577 switch ((channelData->fx[3] >> 4) & 3) {
578 case 0:
579 case 3: // 3 is technically random
580 mutation = MOD_SineTable[mutation];
581 break;
582 case 1:
583 if (channelData->fx[0] & 0x80) {
584 mutation *= -8;
585 mutation += 0xff;
586 } else {
587 mutation *= 8;
588 }
589 break;
590 case 2:
591 mutation = 0xff;
592 break;
593 }
594 mutation *= channelData->fx[2] >> 4;
595 mutation >>= 6;
596 volume = channelData->volume;
597 if (channelData->fx[0] & 0x80) {
598 volume -= mutation;
599 } else {
600 volume += mutation;
601 }
602 newValue = channelData->fx[0] + (channelData->fx[2] & 0x0f);
603 if (newValue >= 32) newValue -= 64;
604 channelData->fx[0] = newValue;
605 if (volume < 0) volume = 0;
606 if (volume > 63) volume = 63;
607 SETVOICEVOLUME(channel, volume);
608 break;
609 case 10: // volume slide
610 volume = channelData->volume;
611 if (effectNibble23 < 0x10) {
612 volume -= effectNibble23;
613 if (volume < 0) volume = 0;
614 } else {
615 volume += effectNibble3;
616 if (volume > 63) volume = 63;
617 }
618 channelData->volume = volume;
619 SETVOICEVOLUME(channel, volume);
620 break;
621 case 14: // extended
622 switch (effectNibble3) {
623 case 9: // retrigger sample
624 // this doesn't look right, we probably want to reset the sample location
625 // Tick 0 only gets here on the repeats of a delayed row. Like ProTracker, a row
626 // with a note doesn't retrigger on tick 0, even then.
627 if ((MOD_Tick == 0) && (((rowPointer[0] & 0x0f) << 8) | rowPointer[1])) break;
628 if (effectNibble2 && (MOD_Tick % effectNibble2) == 0) {
629 SPUKeyOnInstrument(channel, channelData->sampleID);
630 channelData->keyOns++;
631 }
632 break;
633 case 12: // cut sample
634 if (MOD_Tick != effectNibble2) break;
635 channelData->volume = 0;
636 SPUSetVoiceVolume(channel, 0, 0);
637 }
638 break;
639 }
640
641 rowPointer += 4;
642 }

645static void MOD_UpdateRow() {
646 const unsigned channels = MOD_Channels;
647 if (MOD_ChangeOrderNextTick) {
648 unsigned newOrder = MOD_NextOrder;
649 if (newOrder >= MOD_SongLength) newOrder = 0;
650 MOD_CurrentRow = 0;
651 MOD_CurrentOrder = newOrder;
652 MOD_CurrentPattern = MOD_ModuleData[newOrder];
653 }
654 if (MOD_ChangeRowNextTick) {
655 unsigned newRow = (MOD_NextRow >> 4) * 10 + (MOD_NextRow & 0x0f);
656 if (newRow >= 64) newRow = 0;
657 MOD_CurrentRow = newRow;
658 if (MOD_ChangeOrderNextTick) {
659 if (++MOD_CurrentOrder >= MOD_SongLength) MOD_CurrentOrder = 0;
660 MOD_CurrentPattern = MOD_ModuleData[MOD_CurrentOrder];
661 }
662 }
663 MOD_ChangeRowNextTick = 0;
664 MOD_ChangeOrderNextTick = 0;
665 MOD_RowPointer =
666 MOD_ModuleData + 128 + 4 + MOD_CurrentPattern * MOD_Channels * 0x100 + MOD_CurrentRow * channels * 4;
667 const uint8_t* rowPointer = MOD_RowPointer;
668
669 for (unsigned channel = 0; channel < channels; channel++) {
670 int16_t volume;
671 struct SPUChannelData* const channelData = &s_channelData[channel];
672
673 uint8_t effectNibble1 = rowPointer[2];
674 uint8_t effectNibble23 = rowPointer[3];
675 uint16_t nibble0 = rowPointer[0];
676 unsigned sampleID = (nibble0 & 0xf0) | (effectNibble1 >> 4);
677 uint8_t effectNibble2 = effectNibble23 & 0x0f;
678 uint8_t effectNibble3 = effectNibble23 >> 4;
679 unsigned period = ((nibble0 & 0x0f) << 8) | rowPointer[1];
680 int32_t newPeriod;
681 uint8_t fx;
682 effectNibble1 &= 0x0f;
683
684 // samplePos is where the channel's sample starts, like ProTracker's n_start: a sample number
685 // resets it, and 9xx (xx * 256 MOD sample bytes, 900 reusing the channel's last offset) adds
686 // to it, so notes without a sample number keep starting there. It needs to be known before
687 // setting the start address below, which the original code did after, in the effects switch.
688 if (sampleID != 0) channelData->samplePos = 0;
689 if (effectNibble1 == 9) {
690 if (effectNibble23 != 0) channelData->sampleOffset = effectNibble23 << 8;
691 channelData->samplePos += channelData->sampleOffset;
692 }
693 if (sampleID != 0) {
694 channelData->sampleID = --sampleID;
695 volume = s_spuInstrumentData[sampleID].volume;
696 if (volume > 63) volume = 63;
697 channelData->volume = volume;
698 if (effectNibble1 != 7) {
699 SETVOICEVOLUME(channel, volume);
700 }
701 SPUSetStartAddress(channel, SPUInstrumentAddress(sampleID, channelData->samplePos));
702 } else if (effectNibble1 == 9) {
703 // 9xx without a sample number applies to the channel's current sample.
704 SPUSetStartAddress(channel, SPUInstrumentAddress(channelData->sampleID, channelData->samplePos));
705 }
706
707 if (period != 0) {
708 int periodIndex;
709 // Scan the whole finetune 0 row, down to C-1 at index 0. A period that
710 // isn't in it leaves periodIndex at -1.
711 for (periodIndex = 35; periodIndex >= 0; periodIndex--) {
712 if (MOD_PeriodTable[periodIndex] == period) break;
713 }
714 channelData->note = periodIndex + s_spuInstrumentData[channelData->sampleID].finetune * 36;
715 fx = channelData->fx[3];
716 if ((fx & 0x0f) < 4) {
717 channelData->vibrato = 0;
718 }
719 if ((fx >> 4) < 4) {
720 channelData->fx[0] = 0;
721 }
722 if ((effectNibble1 != 3) && (effectNibble1 != 5)) {
723 SPUWaitIdle();
724 SPUKeyOnInstrument(channel, channelData->sampleID);
725 channelData->keyOns++;
726 channelData->period = MOD_PeriodTable[channelData->note];
727 }
728 newPeriod = channelData->period;
729 SETVOICESAMPLERATE(channel, newPeriod);
730 }
731
732 switch (effectNibble1) {
733 case 3: // glissando
734 if (effectNibble23 != 0) {
735 channelData->slideSpeed = effectNibble23;
736 }
737 if (period != 0) {
738 channelData->slideTo = MOD_PeriodTable[channelData->note];
739 }
740 break;
741 case 4: // vibrato
742 if (effectNibble3 != 0) {
743 fx = channelData->fx[1];
744 fx &= ~0x0f;
745 fx |= effectNibble3;
746 channelData->fx[1] = fx;
747 }
748 if (effectNibble2 != 0) {
749 fx = channelData->fx[1];
750 fx &= ~0xf0;
751 fx |= effectNibble2 << 4;
752 channelData->fx[1] = fx;
753 }
754 break;
755 case 7: // tremolo
756 if (effectNibble3 != 0) {
757 fx = channelData->fx[2];
758 fx &= ~0x0f;
759 fx |= effectNibble3;
760 channelData->fx[2] = fx;
761 }
762 if (effectNibble2 != 0) {
763 fx = channelData->fx[2];
764 fx &= ~0xf0;
765 fx |= effectNibble2 << 4;
766 channelData->fx[2] = fx;
767 }
768 break;
769 case 9: // sample jump, handled above
770 break;
771 case 11: // order jump
772 if (!MOD_ChangeOrderNextTick) {
773 MOD_ChangeOrderNextTick = 1;
774 MOD_NextOrder = effectNibble23;
775 }
776 break;
777 case 12: // set volume
778 volume = effectNibble23;
779 if (volume > 64) volume = 63;
780 channelData->volume = volume;
781 SETVOICEVOLUME(channel, volume);
782 break;
783 case 13: // pattern break
784 if (!MOD_ChangeRowNextTick) {
785 MOD_ChangeRowNextTick = 1;
786 MOD_NextRow = effectNibble23;
787 }
788 break;
789 case 14: // extended
790 switch (effectNibble3) {
791 case 1: // fineslide up
792 newPeriod = channelData->period;
793 newPeriod -= effectNibble2;
794 channelData->period = newPeriod;
795 SETVOICESAMPLERATE(channel, newPeriod);
796 break;
797 case 2: // fineslide down
798 newPeriod = channelData->period;
799 newPeriod += effectNibble2;
800 channelData->period = newPeriod;
801 SETVOICESAMPLERATE(channel, newPeriod);
802 break;
803 case 4: // set vibrato waveform
804 fx = channelData->fx[3];
805 fx &= ~0x0f;
806 fx |= effectNibble2;
807 channelData->fx[3] = fx;
808 break;
809 case 5: // set finetune value
810 // Applies to the channel's current sample, which a sample number
811 // on this row has already selected.
812 s_spuInstrumentData[channelData->sampleID].finetune = effectNibble2;
813 break;
814 case 6: // loop pattern
815 if (MOD_LoopCount-- == 0) {
816 MOD_LoopCount = effectNibble2;
817 }
818 if (MOD_LoopCount != 0) {
819 MOD_CurrentRow = MOD_LoopStart;
820 }
821 break;
822 case 7: // set tremolo waveform
823 fx = channelData->fx[3];
824 fx &= ~0xf0;
825 fx |= effectNibble2 << 4;
826 channelData->fx[3] = fx;
827 break;
828 case 9: // retrigger sample
829 // Like ProTracker, a row without a note retriggers on tick 0 too. With a note,
830 // the note itself is the tick 0 trigger. Later ticks are in MOD_UpdateEffect.
831 if (effectNibble2 && (period == 0)) {
832 SPUKeyOnInstrument(channel, channelData->sampleID);
833 channelData->keyOns++;
834 }
835 break;
836 case 10: // fine volume up
837 volume = channelData->volume;
838 volume += effectNibble2;
839 if (volume > 63) volume = 63;
840 channelData->volume = volume;
841 SETVOICEVOLUME(channel, volume);
842 break;
843 case 11: // fine volume down
844 volume = channelData->volume;
845 volume -= effectNibble2;
846 if (volume < 0) volume = 0;
847 channelData->volume = volume;
848 SETVOICEVOLUME(channel, volume);
849 break;
850 case 14: // delay pattern
851 MOD_PatternDelay = effectNibble2;
852 break;
853 }
854 break;
855 case 15: // set speed
856 // the original code here is very wrong with regards to
857 // how to interpret the command; also it was very opinionated
858 // about using timer1 for its clock source
859 if (effectNibble23 == 0) break;
860 if (effectNibble23 < 32) {
861 MOD_Speed = effectNibble23;
862 } else {
863 MOD_SetBPM(effectNibble23);
864 }
865 break;
866 }
867
868 rowPointer += 4;
869 }
870}

873 // the original code is getting the delay pattern wrong here, and
874 // isn't processing them as actual line delays, rather as a sort
875 // of ticks delay, and was basically going too fast
876 uint8_t newPatternDelay = MOD_PatternDelay;
877 if (++MOD_Tick < MOD_Speed) {
878 MOD_UpdateEffect();

880 MOD_Tick = 0;

883 newPatternDelay = MOD_PatternDelay;
