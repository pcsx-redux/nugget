71 ramsyscall_printf("Masked Nop, HINTMSK %02x: irq %i HINTSTS %02x HSTS %02x while masked; irq %i HINTSTS %02x after unmask; response size %i stat %02x\n",
72 masks[i], irqMasked, stsMasked, ctrlMasked, irqAfter, stsAfter, size, stat);
73 // The cause lands in HINTSTS either way; HINTMSK is ANDed with the cause value to gate
74 // the interrupt line, and unmasking raises it. Measured on a SCPH-9002.
