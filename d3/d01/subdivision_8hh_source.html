101[[nodiscard]] constexpr TexturedVertex half(const TexturedVertex& a, const TexturedVertex& b) {
102 return TexturedVertex{

137template <typename Decide, typename EmitQuad, typename EmitFill>
138void divideQuad(const TexturedVertex& a, const TexturedVertex& b, const TexturedVertex& c,
139 const TexturedVertex& d, unsigned depth, unsigned edges, Decide&& decide, EmitQuad&& emitQuad,
140 EmitFill&& emitFill) {
141 if ((depth == 0) || !decide(a, b, c, d, depth)) {
142 emitQuad(a, b, c, d);
143 return;
144 }
145
146 const auto ab = half(a, b);
147 const auto cd = half(c, d);
148 const auto ac = half(a, c);
149 const auto bd = half(b, d);
150 const auto center = half(ab, cd);
151
152 // Every edge of this quad that borders something which may not be divided
153 // is about to gain a vertex the neighbour does not have. Cover each one.
154 if (edges & EdgeAB) emitFill(a, ab, b);
155 if (edges & EdgeBD) emitFill(b, bd, d);
156 if (edges & EdgeCD) emitFill(c, cd, d);
157 if (edges & EdgeAC) emitFill(a, ac, c);
158
159 const unsigned next = depth - 1;
160 divideQuad(a, ab, ac, center, next, edges & (EdgeAB | EdgeAC), decide, emitQuad, emitFill);
161 divideQuad(ab, b, center, bd, next, edges & (EdgeAB | EdgeBD), decide, emitQuad, emitFill);
162 divideQuad(ac, center, c, cd, next, edges & (EdgeAC | EdgeCD), decide, emitQuad, emitFill);
163 divideQuad(center, bd, cd, d, next, edges & (EdgeBD | EdgeCD), decide, emitQuad, emitFill);
164}
