27/* Random well-formed LZ4 blocks. The raw data is built alongside the
28 stream, so the expected output never comes from a decoder. Every block
29 obeys the end-of-block rules: the last 5 bytes are literals, the last match
30 starts at least 12 bytes before the end, and a block with no match is all
31 literals. Lengths lean on the encoding edges (14-17, 268-271, 523-526) and
32 offsets on 1-8 and the farthest one available. 32-bit arithmetic only.
33 Define GEN_RAW and GEN_COMP (byte arrays of GEN_MAX bytes) before including. */
34
35#include <stdint.h>
36
37#define GEN_MAX (192 * 1024)
38
39/* Initial xorshift states, seed * 0x9e3779b97f4a7c15 + 1, seeds 1 to 5. */
40static const uint32_t s_genSeeds[6][2] = {
41 {0x00000001u, 0x00000000u}, {0x7f4a7c16u, 0x9e3779b9u}, {0xfe94f82bu, 0x3c6ef372u},
42 {0x7ddf7440u, 0xdaa66d2cu}, {0xfd29f055u, 0x78dde6e5u}, {0x7c746c6au, 0x1715609fu},
43};

45static uint32_t s_genLo, s_genHi;
46static uint32_t s_rawlen, s_complen;
47
48static void genSeed(int seed) {
49 s_genLo = s_genSeeds[seed][0];
50 s_genHi = s_genSeeds[seed][1];
51}

53/* 64-bit xorshift (13, 7, 17) on a hi:lo pair; returns bits 16..47. */
54static uint32_t rnd(void) {
55 uint32_t lo = s_genLo, hi = s_genHi;
56 hi ^= (hi << 13) | (lo >> 19);
57 lo ^= lo << 13;
58 lo ^= (lo >> 7) | (hi << 25);
59 hi ^= hi >> 7;
60 hi ^= (hi << 17) | (lo >> 15);
61 lo ^= lo << 17;
62 s_genLo = lo;
63 s_genHi = hi;
64 return (lo >> 16) | (hi << 16);
65}

67static uint32_t pick(uint32_t n) { return n ? rnd() % n : 0; }
68
69/* Length biased toward the encoding edges. */
70static uint32_t len(uint32_t base) {
71 static const uint32_t edges[] = {0, 1, 2, 3, 13, 14, 15, 16, 17, 268, 269, 270, 271, 523, 524, 525, 526, 1000};
72 switch (pick(4)) {
73 case 0: {
74 uint32_t e = edges[pick(sizeof(edges) / sizeof(edges[0]))];
75 return e >= base ? e - base : 0;
76 }
77 case 1:
78 return pick(8);
79 case 2:
80 return pick(64);
81 default:
82 return pick(2000);
83 }
84}

86static void emitLen(uint32_t n) {
87 while (n >= 255) {
88 GEN_COMP[s_complen++] = 255;
89 n -= 255;
90 }
91 GEN_COMP[s_complen++] = n;
92}

94static void gen(void) {
95 s_rawlen = s_complen = 0;
96 uint32_t nseq = pick(4) == 0 ? 0 : pick(40);
97 for (uint32_t q = 0; q < nseq; q++) {
98 uint32_t lit = len(0);
99 if (s_rawlen + lit == 0) lit = 1 + pick(20);
100 uint32_t ml = 4 + len(4);
101 uint32_t token = s_complen++;
102 if (lit >= 15) emitLen(lit - 15);
103 int fill = pick(3);
104 for (uint32_t i = 0; i < lit; i++) {
105 uint8_t b = fill == 0 ? rnd() : fill == 1 ? 'a' + pick(4) : 0;
106 GEN_COMP[s_complen++] = b;
107 GEN_RAW[s_rawlen++] = b;
108 }
109 uint32_t maxoff = s_rawlen > 65535 ? 65535 : s_rawlen;
110 uint32_t off;
111 switch (pick(4)) {
112 case 0:
113 off = 1 + pick(maxoff < 8 ? maxoff : 8);
114 break;
115 case 1:
116 off = maxoff;
117 break;
118 default:
119 off = 1 + pick(maxoff);
120 }
121 GEN_COMP[s_complen++] = off & 0xff;
122 GEN_COMP[s_complen++] = off >> 8;
123 uint32_t mn = ml - 4;
124 if (mn >= 15) emitLen(mn - 15);
125 GEN_COMP[token] = (uint8_t)(((lit < 15 ? lit : 15) << 4) | (mn < 15 ? mn : 15));
126 for (uint32_t i = 0; i < ml; i++, s_rawlen++) GEN_RAW[s_rawlen] = GEN_RAW[s_rawlen - off];
127 }
128 /* A trailer of >= 12 literals after any match keeps the last match
129 starting at least 12 bytes before the end. */
130 uint32_t tail = 5 + len(5);
131 if (nseq && tail < 12) tail = 12;
132 uint32_t token = s_complen++;
133 if (tail >= 15) emitLen(tail - 15);
134 GEN_COMP[token] = (uint8_t)((tail < 15 ? tail : 15) << 4);
135 for (uint32_t i = 0; i < tail; i++) {
136 uint8_t b = rnd();
137 GEN_COMP[s_complen++] = b;
138 GEN_RAW[s_rawlen++] = b;
139 }
140}
