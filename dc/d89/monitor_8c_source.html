36/* READ_MEM responses stream out of target memory in 8 KiB chunks (design
37 section 13). Nothing is staged: bulk data goes straight to/from the operation's
38 real address, so the only buffer the monitor needs is a tiny one for the small
39 fixed-size commands (RUN carries the most at 6 words). */
40#define MON_CHUNK_BYTES 8192
41#define MON_CMD_WORDS 16

45/* Saved context of the halted program (== the current thread's register frame,

52/* Running DCIC image + the armed data-watch address, so SET_BP/CLR_BP can

57/* SR the target runs under after a fresh RUN. The rfe in returnFromException
58 pops IEp->IEc, so IEp (bit2) enables interrupts; IM2 (bit10) unmasks the PS1

112/* READ_MEM [addr:u32][len:u32] -> one or more DATA [nbytes:u32][bytes] frames,
113 each capped at an 8 KiB chunk; the host concatenates until it has len bytes. */
114static void cmdReadMem(const uint16_t *p) {
115 uint32_t addr = rd32(p, 0);
116 uint32_t len = rd32(p, 2);
117 volatile const uint8_t *src = (volatile const uint8_t *)addr;
118
119 uint32_t off = 0;
120 do {
121 uint32_t chunk = len - off;
122 if (chunk > MON_CHUNK_BYTES) chunk = MON_CHUNK_BYTES;
123 uint32_t words = (chunk + 1) >> 1;
124
125 transportSendBegin(MON_DATA, (uint16_t)(2 + words));
126 sendWord32(chunk); /* nbytes in THIS frame */
127 for (uint32_t i = 0; i < words; i++) {
128 uint32_t bi = off + 2 * i;
129 uint8_t b0 = src[bi];
130 uint8_t b1 = (2 * i + 1 < chunk) ? src[bi + 1] : 0;
131 transportSendWord((uint16_t)(b0 | (b1 << 8)));
132 }
133 transportSendEnd();
134 off += chunk;
135 } while (off < len);
136}

138static uint32_t recvU32(void) {
139 uint32_t lo = transportRecvWord();
140 uint32_t hi = transportRecvWord();
141 return lo | (hi << 16);
142}

144/* WRITE_MEM / LOAD [addr:u32][len:u32][bytes] -> ACK | ERROR. Called mid-frame:
145 transportRecvBegin has already consumed TYPE and LEN. The bytes are decoded
146 straight into the target address as they arrive - no staging buffer - so a
147 single frame carries an arbitrarily large write (host still chunks at 8 KiB).
148 `frameWords` is the payload word count from the frame header. */
149static void streamWriteMem(uint16_t frameWords) {
150 uint32_t addr = recvU32(); /* 2 words */
151 uint32_t nbytes = recvU32(); /* 2 words */
152 uint32_t consumed = 4;
153 volatile uint8_t *dst = (volatile uint8_t *)addr;
154
155 for (uint16_t w = consumed; w < frameWords; w++) {
156 uint16_t word = transportRecvWord();
157 uint32_t bi = (uint32_t)(w - consumed) * 2;
158 if (bi < nbytes) dst[bi] = (uint8_t)(word & 0xff);
159 if (bi + 1 < nbytes) dst[bi + 1] = (uint8_t)(word >> 8);
160 }
161
162 int rc = transportRecvEnd();
163 if (rc == TRANSPORT_OK) {
164 sendAck();
165 } else {
166 sendError(MON_ECKSUM);
167 }
168}

170/* GET_REGS [] -> REGS [38 x u32, gdb g-packet order]. */
171static void cmdGetRegs(void) {
172 struct Registers *r = s_ctx;
173 transportSendBegin(MON_REGS, 38 * 2);
174 for (int i = 0; i < 32; i++) sendWord32(r ? r->GPR.r[i] : 0); /* 0..31 r0..r31 */
175 sendWord32(r ? r->SR : 0); /* 32 SR */
176 sendWord32(r ? r->lo : 0); /* 33 LO */
177 sendWord32(r ? r->hi : 0); /* 34 HI */
178 sendWord32(s_badVaddr); /* 35 BadVaddr */
179 sendWord32(r ? r->Cause : 0); /* 36 Cause */
180 sendWord32(r ? r->returnPC : 0); /* 37 PC (EPC) */
181 transportSendEnd();
182}

184/* SET_REG [idx:u16][value:u32] -> ACK | ERROR. idx indexes the REGS table. */
185static void cmdSetReg(const uint16_t *p) {
186 uint16_t idx = p[0];
187 uint32_t val = rd32(p, 1);
188 struct Registers *r = s_ctx;
189
190 if (idx > 37) {
191 sendError(MON_EBADREG);
192 return;
193 }
194 if (!r) {
195 sendError(MON_EBADSTATE);
196 return;
197 }
198
199 if (idx < 32) {
200 if (idx != 0) r->GPR.r[idx] = val; /* r0 stays 0 */
201 } else {
202 switch (idx) {
203 case 32: r->SR = val; break;
204 case 33: r->lo = val; break;
205 case 34: r->hi = val; break;
206 case 35: s_badVaddr = val; break;
207 case 36: r->Cause = val; break;
208 case 37: r->returnPC = val; break;
209 }
210 }
211 sendAck();
212}

214/* SET_BP [kind:u16][addr:u32][mask:u32] -> ACK.
215 kind: 0 exec, 1 data-read, 2 data-write, 3 data-rw. */
216static void cmdSetBp(const uint16_t *p) {
217 uint16_t kind = p[0];
218 uint32_t addr = rd32(p, 1);
219 uint32_t mask = rd32(p, 3);
220
221 if (kind == 0) {
222 writeBPC(addr);
223 writeBPCM(mask);
224 s_dcic |= DCIC_DE | DCIC_PCE | DCIC_TR | DCIC_KD | DCIC_UD;
225 } else if (kind <= 3) {
226 writeBDA(addr);
227 writeBDAM(mask);
228 s_watchAddr = addr;
229 uint32_t enables = DCIC_DE | DCIC_DAE | DCIC_TR | DCIC_KD | DCIC_UD;
230 if (kind == 1) enables |= DCIC_DR;
231 if (kind == 2) enables |= DCIC_DW;
232 if (kind == 3) enables |= DCIC_DR | DCIC_DW;
233 s_dcic |= enables;
234 } else {
235 sendError(MON_EBADCMD);
236 return;
237 }
238 writeDCIC(s_dcic);
239 sendAck();
240}
241
242/* CLR_BP [kind:u16] -> ACK. */
243static void cmdClrBp(const uint16_t *p) {
244 uint16_t kind = p[0];
245 if (kind == 0) {
246 s_dcic &= ~DCIC_PCE;
247 } else {
248 s_dcic &= ~(DCIC_DAE | DCIC_DR | DCIC_DW);
249 }
250 if ((s_dcic & (DCIC_PCE | DCIC_DAE)) == 0) s_dcic = 0; /* drop master enable */
251 writeDCIC(s_dcic);
252 sendAck();
253}

255/* RUN [pc:u32][gp:u32][sp:u32] -> ACK, then hand the CPU to the target. Never
256 returns (returnFromException does an rfe into pc). */
257static void cmdRun(const uint16_t *p) {
258 uint32_t pc = rd32(p, 0);
259 uint32_t gp = rd32(p, 2);
260 uint32_t sp = rd32(p, 4);
261 struct Registers *r = currentRegs();
262
263 for (int i = 0; i < 32; i++) r->GPR.r[i] = 0;
264 r->GPR.n.gp = gp;
265 r->GPR.n.sp = sp;
266 r->GPR.n.fp = sp;
267 r->returnPC = pc;
268 r->hi = 0;
269 r->lo = 0;
270 r->SR = MON_RUN_SR;
271 r->Cause = 0;
272 s_ctx = 0; /* running: no halted context */
273
274 sendAck();
275 monitorResume();
276}

288/* SET_BAUD [reload:u16] -> ACK at the old rate, then PONG at the new one if
289 the host's PING arrives there (design section 2a). */
290static void cmdSetBaud(const uint16_t *p) {
291 uint16_t reload = p[0];
292 if (reload == 0) {
293 sendError(MON_EBADLEN);
294 return;
295 }
296 if (!transportHasRate()) {
297 sendError(MON_EBADCMD);
298 return;
299 }
300 sendAck();
301 uint16_t ver = MON_PROTO_VER;
302 if (transportTryRate(reload, MON_PONG, ver) == 1) transportSendFrame(MON_PONG, &ver, 1);
303}

305/* Dispatch one small HALTED-state command whose payload is already buffered in
306 s_cmd. WRITE_MEM/LOAD are NOT here - they stream directly to target memory in
307 the loop. RUN/CONT do not return (they resume the target). */
308static void dispatchCommand(uint16_t type, const uint16_t *payload) {
309 switch (type) {
310 case MON_PING: {
311 uint16_t ver = MON_PROTO_VER;
312 transportSendFrame(MON_PONG, &ver, 1);
313 break;
314 }
315 case MON_READ_MEM: cmdReadMem(payload); break;
316 case MON_GET_REGS: cmdGetRegs(); break;
317 case MON_SET_REG: cmdSetReg(payload); break;
318 case MON_SET_BP: cmdSetBp(payload); break;
319 case MON_CLR_BP: cmdClrBp(payload); break;
320 case MON_RUN: cmdRun(payload); break;
321 case MON_CONT: cmdCont(); break;
322 case MON_SET_BAUD: cmdSetBaud(payload); break;
323 case MON_STOP:
324 /* Only meaningful while RUNNING, where it is serviced by the
325 interrupt-poll path (v1 seam, below). In HALTED it is a no-op. */
326 sendAck();
327 break;
328 default: sendError(MON_EBADCMD); break;
329 }
330}

332/* Blocking HALTED command loop. Bulk WRITE_MEM/LOAD payloads are decoded
333 straight into target memory (no staging); every other command has a small
334 fixed payload buffered in s_cmd. Returns only via a resume, which is noreturn,
335 so in practice it does not return. */
336static __attribute__((noreturn)) void monitorCommandLoop(void) {
337 for (;;) {
338 uint16_t type;
339 uint16_t len;
340 transportRecvBegin(&type, &len);
341
342 if (type == MON_WRITE_MEM || type == MON_LOAD) {
343 streamWriteMem(len); /* + LOAD extent bookkeeping later */
344 continue;
345 }
346
347 uint16_t n = (len <= MON_CMD_WORDS) ? len : MON_CMD_WORDS;
348 for (uint16_t i = 0; i < n; i++) s_cmd[i] = transportRecvWord();
349 for (uint16_t i = n; i < len; i++) transportRecvWord(); /* drain overflow */
350 int rc = transportRecvEnd();
351
352 if (rc != TRANSPORT_OK) {
353 sendError(MON_ECKSUM);
354 continue;
355 }
356 if (len > MON_CMD_WORDS) {
357 sendError(MON_EBADLEN);
358 continue;
359 }
360 dispatchCommand(type, s_cmd);
361 }
362}

364/* Snapshot a stop, tell the host, and drop into the command loop until the host
365 resumes. Never returns. */
366static __attribute__((noreturn)) void monitorStop(struct Registers *r, uint16_t reason, uint32_t a, uint32_t b) {
367 s_ctx = r;
368 s_badVaddr = (reason == MON_STOP_FAULT) ? b : 0;
369 emitStopped(reason, r->returnPC, a, b);
370 monitorCommandLoop();
371 __builtin_unreachable();
372}

376/* Priority-0 verifier, prepended ahead of the kernel's syscall verifier so it
377 sees break (ExcCode 9) and faults first. It reuses the kernel's context save
378 (the frame is already in currentRegs()); both the 0x80 general vector and the
379 0x40 cop0-break vector route here. Returns 0 for exceptions it does not own
380 (interrupt, syscall) so the rest of the chain runs; when it owns one it never
381 returns (it resumes the target or enters the command loop). */
382static int monitorVerifier(void) {
383 struct Registers *r = currentRegs();
384 uint32_t excode = CAUSE_EXCCODE(r->Cause);
385
386 if (excode == EXCCODE_BP) {
387 uint32_t insn = *(volatile uint32_t *)(r->returnPC);
388 if ((insn & 0x3f) == 0x0d) {
389 /* Software `break`: decode break code1, code2 (design section 10). */
390 uint32_t code1 = (insn >> 16) & 0x3ff;
391 uint32_t code2 = (insn >> 6) & 0x3ff;
392 if (code1 == 0 && code2 >= 0x101 && code2 <= 0x107) {
393 monitorServicePcdrv(r, code2);
394 r->returnPC += 4; /* step past the break; program stays RUNNING */
395 monitorResume();
396 }
397 if (code1 == 4) {
398 switch (code2) {
399 case 0: /* exit break (break 4, 0): exit code in $a0. */
400 monitorStop(r, MON_STOP_EXIT, r->GPR.n.a0, 0);
401 break;
402 case 1: /* enter the monitor command loop from (break 4, 1). */
403 emitHello();
404 monitorCommandLoop();
405 break;
406 }
407 }
408 /* Any other software break -> generic debugger stop. Advance past it
409 so a following CONT resumes the instruction after the break. */
410 r->returnPC += 4;
411 monitorStop(r, MON_STOP_BREAKPOINT, 0, 0);
412 } else {
413 /* No `break` at EPC: a cop0 hardware breakpoint. Disable the debug
414 unit first (design section 10), then report. */
415 uint32_t dcic = readDCIC();
416 writeDCIC(0);
417 s_dcic = 0;
418 if (dcic & DCIC_DAE) {
419 monitorStop(r, MON_STOP_DATA_WATCH, s_watchAddr, 0);
420 } else {
421 monitorStop(r, MON_STOP_BREAKPOINT, 0, 0);
422 }
423 }
424 }
425
426 if (excode == EXCCODE_ADEL || excode == EXCCODE_ADES) {
427 monitorStop(r, MON_STOP_FAULT, excode, readBadVaddr());
428 }
429 if (excode == EXCCODE_IBE || excode == EXCCODE_DBE || excode == EXCCODE_RI || excode == EXCCODE_CPU ||
430 excode == EXCCODE_OVF) {
431 monitorStop(r, MON_STOP_FAULT, excode, 0);
432 }
433
434 /* Interrupt (0) / syscall (8) / anything else: not ours. The async STOP
435 (host Ctrl-C) poll-on-any-interrupt is a v1 seam - it would peek the word
436 channel here and, if a STOP frame is pending, snapshot and halt. Banked
437 for now (design section 12); the target must be interruptible to use it. */
438 return 0;
439}

441/* Copy the installed 0x80 general-exception trampoline down to the 0x40 cop0
442 break vector so a hardware breakpoint routes through the same handler and
443 chain. Nothing is installed at 0x40 by OpenBIOS. */
444static void installCop0BreakVector(void) {
445 volatile uint32_t *v80 = (volatile uint32_t *)0x80;
446 volatile uint32_t *v40 = (volatile uint32_t *)0x40;
447 for (int i = 0; i < 4; i++) v40[i] = v80[i];
448 syscall_flushCache();
449}

463 s_dcic = 0;
