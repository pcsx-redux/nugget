30 // Waits for one IRQ, up to timeoutUs. Returns its cause, or 0 if none came.
31 static uint8_t sqWait(uint32_t timeoutUs, uint8_t response[16], uint8_t* size) {
32 uint32_t timeout = timeoutUs;
33 if (!waitCDRomIRQWithTimeout(&timeout)) {
34 *size = 0;
35 return 0;
36 }
37 uint8_t cause = ackCDRomCause();
38 *size = readResponse(response);
39 return cause;
40 }
