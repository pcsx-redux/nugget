31// A FastTracker 2 module player for the SPU. Songs are converted with the xmconv tool from
32// PCSX-Redux, which writes two files: the song (patterns, instruments, sample headers), which stays
33// in main RAM while playing, and the samples, which are uploaded to SPU RAM by XM_Load and can be
34// freed afterwards.
35//
36// Each XM channel plays on the SPU voice with the same number. Voices from XM_Channels to 23 are
37// left alone while the song plays, and can be used for sound effects with XM_PlaySoundEffect.
38
39// Song file format, version 1. All fields are little endian. Offsets are from the start of the file.
40struct XMSongHeader {
41 char magic[4]; // "PSXM"
42 uint16_t version;
43 uint16_t headerSize;
44 uint8_t channels;
45 uint8_t flags; // bit 0: linear frequency table, otherwise Amiga periods
46 uint8_t speed; // initial ticks per row
47 uint8_t bpm; // initial tempo
48 uint16_t songLength;
49 uint16_t restart;
50 uint16_t numPatterns;
51 uint16_t numInstruments;
52 uint16_t numSamples;
53 // 0 for FastTracker 2 Amiga periods. Otherwise ProTracker periods (MOD files): 16 rows, finetune
54 // -8 to 7, of 36 uint16_t periods times 4, C-1 to B-3 in ProTracker octaves, at this offset; notes
55 // outside that range halve or double the nearest octave, and every period is held to 113..856.
56 uint16_t periodTableOffset;
57 uint32_t sampleBytes; // size of the sample file
58 uint32_t ordersOffset; // songLength bytes, pattern numbers
59 uint32_t patternsOffset; // numPatterns of struct XMPatternEntry
60 uint32_t instrumentsOffset; // numInstruments of struct XMInstrument
61 uint32_t samplesOffset; // numSamples of struct XMSample
62 char title[20];
63};
64
65// Pattern data uses the XM packing: a byte with bit 7 set is a mask of which of note, instrument,
66// volume, effect and parameter follow; otherwise that byte is the note, and the four others follow.
67struct XMPatternEntry {
68 uint32_t offset;
69 uint16_t rows;
70 uint16_t size;
71};
72
73struct XMEnvelope {
74 uint8_t count, sustain, loopStart, loopEnd, flags; // flags: bit 0 on, bit 1 sustain, bit 2 loop
75};
76
77struct XMInstrument {
78 uint8_t keymap[96]; // sample index for each note, 0xff for none
79 uint16_t volPoints[12][2]; // tick, value 0..64
80 uint16_t panPoints[12][2]; // tick, value 0..64
81 struct XMEnvelope vol, pan; // 10 bytes
82 uint8_t vibType, vibSweep, vibDepth, vibRate;
83 uint16_t fadeout;
84};
85
86struct XMSample {
87 uint32_t spuOffset; // bytes into the sample file
88 uint32_t spuLength; // bytes
89 uint32_t length; // samples, at the converted rate
90 uint32_t loopStart; // samples, at the converted rate
91 uint32_t loopLength; // samples; for a ping-pong loop, the unfolded period
92 uint16_t pad; // silent samples written before the start, to align the loop
93 uint8_t flags; // bit 0 loop, bit 1 ping-pong
94 uint8_t rateShift; // the sample was resampled to 1 / (1 << rateShift) of its rate
95 uint8_t volume; // 0..64
96 int8_t finetune;
97 uint8_t pan;
98 int8_t relnote;
99 uint32_t reserved;
100};

112// Number of hblanks to wait before the next call to XM_Poll. It changes from one call to the next, as the
113// fractional part of the tick length is carried over, so read it again after every call.

116// Checks the song file. Returns the number of channels, or 0 if the file isn't a song this player can
117// play.
118unsigned XM_Check(const void* song);
119
120// Loads the song and uploads the samples to SPU RAM, then gets ready to play. The song has to stay in
121// memory, 4-bytes aligned, while it plays; the samples can be freed once this returns. If samples is
122// NULL, they are considered already uploaded. Returns the number of channels, or 0 on error. This
123// resets the SPU and all 24 voices, so sound effects have to be started after it. The upload writes
124// SPU RAM from 0x1010 for sampleBytes rounded up to a multiple of 64 bytes.
125unsigned XM_Load(const void* song, const uint8_t* samples);
126
127// Call this at the rate given by XM_hblanks to play.
128void XM_Poll(void);
129
130// Master music volume, 0 to 65535, defaults to 16384. Not reset by XM_Load.
131void XM_SetMusicVolume(uint32_t volume);
132
133// Plays a note from the song's instruments on a voice the song doesn't use. Instrument is 1-based,
134// note is 1 (C-0) to 96, volume is 0 to 64, and is not affected by the music volume.
135void XM_PlaySoundEffect(unsigned voice, unsigned instrument, unsigned note, unsigned volume);

137// Resets the SPU and silences everything.
138void XM_Silence(void);
