31// A FastTracker 2 module player for the SPU. Songs are converted with the xmconv tool from
32// PCSX-Redux, which writes two files: the song (patterns, instruments, sample headers), which stays
33// in main RAM while playing, and the samples, which are uploaded to SPU RAM by XM_Load and can be
34// freed afterwards.
35//
36// Each XM channel plays on the SPU voice with the same number. Voices from XM_Channels to 23 are
37// left alone while the song plays, and can be used for sound effects with XM_PlaySoundEffect.
38
39// Song file format, version 1 or 2. All fields are little endian. Offsets are from the start of the
40// file. Version 2 adds what the SPU can do and FastTracker 2 cannot say: an exact tick rate,
41// reverb, the hardware ADSR, noise and pitch modulation per sample, and a second, extended effect
42// per cell. A version 1 file plays the same in both.
43struct XMSongHeader {
44 char magic[4]; // "PSXM"
45 uint16_t version;
46 uint16_t headerSize;
47 uint8_t channels;
48 uint8_t flags; // bit 0: linear frequency table, otherwise Amiga periods
49 uint8_t speed; // initial ticks per row
50 uint8_t bpm; // initial tempo
51 uint16_t songLength;
52 uint16_t restart;
53 uint16_t numPatterns;
54 uint16_t numInstruments;
55 uint16_t numSamples;
56 // 0 for FastTracker 2 Amiga periods. Otherwise ProTracker periods (MOD files): 16 rows, finetune
57 // -8 to 7, of 36 uint16_t periods times 4, C-1 to B-3 in ProTracker octaves, at this offset; notes
58 // outside that range halve or double the nearest octave, and every period is held to 113..856.
59 uint16_t periodTableOffset;
60 uint32_t sampleBytes; // size of the sample file
61 uint32_t ordersOffset; // songLength bytes, pattern numbers
62 uint32_t patternsOffset; // numPatterns of struct XMPatternEntry
63 uint32_t instrumentsOffset; // numInstruments of struct XMInstrument
64 uint32_t samplesOffset; // numSamples of struct XMSample
65 char title[20];
66 // version 2 only, headerSize 72
67 uint32_t tickRate; // ticks per second in Q16 at the initial bpm, 0 for bpm * 2 / 5; Fxx scales it;
68 // a tick is at most 65535 hblanks, about 4 s
69 uint32_t reverbOffset; // struct XMReverb, or 0 to leave reverb off
70};

72// Version 2: the reverb unit's setup, written once by XM_Load, which also clears its work area,
73// from base * 8 to the end of SPU RAM. Samples have to end below it.
74struct XMReverb {
75 uint16_t regs[32]; // 0x1f801dc0 onwards, dAPF1 to vRIN
76 uint16_t base; // mBASE, in 8-byte units
77 int16_t volLeft, volRight;
78 uint16_t reserved;
79};

81// Pattern data uses the XM packing: a byte with bit 7 set is a mask of which of note, instrument,
82// volume, effect and parameter follow; otherwise that byte is the note, and the four others follow.
83// Version 2: mask bit 5 adds an extended effect after them, one byte for the effect and two for its
84// little endian parameter, played at the row's first tick:
85// 1 levels: left in the low byte, right in the high byte, 0 to 255. The channel's volume goes to
86// each side scaled by its level, with no pan law, until a pan effect (8xx or a volume column
87// Px) sets panning again
88// 2 noise clock, 0 to 63, SPUCNT bits 13-8
89struct XMPatternEntry {
90 uint32_t offset;
91 uint16_t rows;
92 uint16_t size;
93};

95struct XMEnvelope {
96 uint8_t count, sustain, loopStart, loopEnd, flags; // flags: bit 0 on, bit 1 sustain, bit 2 loop
97};

99struct XMInstrument {
100 uint8_t keymap[96]; // sample index for each note, 0xff for none
101 uint16_t volPoints[12][2]; // tick, value 0..64
102 uint16_t panPoints[12][2]; // tick, value 0..64
103 struct XMEnvelope vol, pan; // 10 bytes
104 uint8_t vibType, vibSweep, vibDepth, vibRate;
105 uint16_t fadeout;
106};

108struct XMSample {
109 uint32_t spuOffset; // bytes into the sample file
110 uint32_t spuLength; // bytes
111 uint32_t length; // samples, at the converted rate
112 uint32_t loopStart; // samples, at the converted rate
113 uint32_t loopLength; // samples; for a ping-pong loop, the unfolded period
114 uint16_t pad; // silent samples written before the start, to align the loop
115 // bit 0 loop, bit 1 ping-pong; version 2: bit 2 hardware ADSR from adsr (a key off releases it,
116 // and the volume envelope and fadeout still apply), bit 3 reverb, bit 4 noise instead of the
117 // sample, bit 5 pitch modulated by the voice below
118 uint8_t flags;
119 uint8_t rateShift; // the sample was resampled to 1 / (1 << rateShift) of its rate
120 uint8_t volume; // 0..64
121 int8_t finetune;
122 uint8_t pan;
123 int8_t relnote;
124 uint32_t adsr; // version 2: ADSR1 in the low half, ADSR2 in the high half; 0 in version 1
125};

141// Checks the song file. Returns the number of channels, or 0 if the file isn't a song this player can

145// Loads the song and uploads the samples to SPU RAM, then gets ready to play. The song has to stay in
146// memory, 4-bytes aligned, while it plays; the samples can be freed once this returns. If samples is
147// NULL, they are considered already uploaded. Returns the number of channels, or 0 on error. This
148// resets the SPU and all 24 voices, so sound effects have to be started after it. The upload writes
149// SPU RAM from 0x1010 for sampleBytes rounded up to a multiple of 64 bytes; a version 2 reverb
150// block also clears SPU RAM from its base to the end.
151unsigned XM_Load(const void* song, const uint8_t* samples);
152
153// Call this at the rate given by XM_hblanks to play.
154void XM_Poll(void);

159// Plays a note from the song's instruments on a voice the song doesn't use. Instrument is 1-based,
160// note is 1 (C-0) to 96, volume is 0 to 64, and is not affected by the music volume.
