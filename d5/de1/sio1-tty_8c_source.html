27/* Kernel TTY on SIO1 for the monitor's retail hosts, after Unirom's
28 ttyredirect: replace the kernel's "tty" device with one whose writes go out
29 the monitor's link as console text, and reopen stdin/stdout on it, so a
30 target's printf reaches the host between frames. A 0x00 byte would enter a
31 frame on a stream link, so it is dropped. Reads return nothing: SIO1 input
32 belongs to the monitor. */
33#include <stdint.h>
34
35#include "common/psxlibc/device.h"
36#include "common/psxlibc/stdio.h"
37#include "common/syscalls/syscalls.h"
38#include "monitor/link.h"

40static int ttyNull(void) { return 0; }
41
42static int ttyAction(struct File *file, enum FileAction action) {
43 if (action != PSXWRITE) return 0;
44 const uint8_t *p = (const uint8_t *)file->buffer;
45 int count = file->count;
46 for (int i = 0; i < count; i++) {
47 if (p[i]) linkPutByte(p[i]);
48 }
49 return count;
50}

75static inline int removeDevice(const char *name) {
76 register int n asm("t1") = 0x48;
77 __asm__ volatile("" : "=r"(n) : "r"(n));
78 return ((int (*)(const char *))0xb0)(name);
79}

81void installSio1Tty(void) {
82 syscall_close(0);
83 syscall_close(1);
84 removeDevice("tty");
85 syscall_addDevice(&s_ttyDevice);
86 syscall_open("tty00:", PSXF_READ);
87 syscall_open("tty00:", PSXF_WRITE);
88}
