40// --------------------------------------------------------------------------
41// 1. GP0(E1) draw-mode fields reflect in GPUSTAT bits 0-10. Field values:
42// texpage X base = 5, texpage Y base 1 = 1, semi-transparency = 2,
43// texture depth = 1, dither = 1, draw-to-display = 1 -> low bits 0x6D5.
44// --------------------------------------------------------------------------
45CESTER_TEST(command_e1_draw_mode_reflects_in_gpustat, gpu_command,
46 rasterReset();
47 cmdSend(0xe10006d5u);
48 ASSERT_STAT_EQ(0x6d5u, 0x7ffu);
